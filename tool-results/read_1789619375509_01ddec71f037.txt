    41→/** Full French date from a local YYYY-MM-DD key — built from local parts, never UTC */
    42→function dueDateLabel(key: string): string {
    43→  const [y, m, d] = key.split("-").map(Number);
    44→  if (!y || !m || !d) return key;
    45→  return frDate(new Date(y, m - 1, d).getTime());
    46→}

    57→/** Compact recurrence selector — « Aucune / Quotid. / Ouvrés / Hebdo / X j » chips (add form + inline edit) */
    58→function RecurrencePicker({
    59→  value,
    60→  onChange,
    61→  label,
    62→  days,
    63→  onDaysChange,
    64→}: {
    65→  value: TaskRecurrence | null;
    66→  onChange: (next: TaskRecurrence | null) => void;
    67→  label: string;
    68→  /** Interval in days for the « custom » recurrence (2–365) */
    69→  days: number;
    70→  onDaysChange: (next: number) => void;
    71→}) {
    72→  /* Raw text while typing — partial input (« 1 » en route to « 12 ») must not be clamped mid-keystroke */
    73→  const [rawDays, setRawDays] = useState<string | null>(null);
    74→
    75→  /* Derived from RECURRENCE_SHORT so every recurrence kind gets its pill automatically */
    76→  const options: Array<{ value: TaskRecurrence | null; label: string }> = [
    77→    { value: null, label: "Aucune" },
    78→    ...(Object.entries(RECURRENCE_SHORT) as Array<[TaskRecurrence, string]>).map(
    79→      ([v, l]) => ({
    80→        value: v,
    81→        // Once selected, the custom pill echoes the live interval (e.g. « 3 j »)
    82→        label: v === "custom" && value === "custom" ? `${days} j` : l,
    83→      }),
    84→    ),
    85→  ];
    86→
    87→  const commitDays = (text: string) => {
    88→    const n = Number.parseInt(text, 10);
    89→    if (Number.isFinite(n)) onDaysChange(clampRecurrenceDays(n));
    90→  };
    91→
    92→  return (
    93→    <div role="group" aria-label={label} className="flex flex-wrap items-center gap-1.5">
    94→      <span aria-hidden className="text-[11px] text-faint">
    95→        Répétition
    96→      </span>
    97→      {options.map((o) => (
    98→        <button
    99→          key={o.value ?? "none"}
   100→          type="button"
   101→          aria-pressed={value === o.value}
   102→          onClick={() => onChange(o.value)}
   103→          className={`rounded-full border px-2.5 py-1 text-xs transition-colors ${
   104→            value === o.value
   105→              ? "border-brand/40 bg-brand/15 font-medium text-brand"
   106→              : "text-muted-foreground hover:text-foreground"
   107→          }`}
   108→        >
   109→          {o.label}
   110→        </button>
   111→      ))}
   112→      {value === "custom" && (
   113→        <label
   114→          title="Répéter tous les N jours (2 à 365)"
   115→          className="inline-flex items-center gap-1 rounded-full border border-brand/40 bg-brand/15 px-2.5 py-1 text-xs font-medium text-brand"
   116→        >
   117→          <span aria-hidden>Tous les</span>
   118→          <input
   119→            type="number"
   120→            inputMode="numeric"
   121→            min={RECURRENCE_DAYS_MIN}
   122→            max={RECURRENCE_DAYS_MAX}
   123→            step={1}
   124→            value={rawDays ?? String(days)}
   125→            aria-label="Nombre de jours entre chaque répétition, entre 2 et 365"
   126→            onChange={(e) => {
   127→              setRawDays(e.target.value);
   128→              commitDays(e.target.value);
   129→            }}
   130→            onBlur={() => setRawDays(null)}
   131→            className="tnum w-11 rounded-full bg-transparent text-center text-xs font-semibold text-brand outline-none [appearance:textfield] focus-visible:ring-2 focus-visible:ring-brand/60 [&::-webkit-inner-spin-button]:appearance-none [&::-webkit-outer-spin-button]:appearance-none"
   132→          />
   133→          <span aria-hidden>jours</span>
   134→        </label>
   135→      )}
   136→    </div>
   137→  );
   138→}

   142→  const addTask = useFocusly((s) => s.addTask);
   143→  const updateTask = useFocusly((s) => s.updateTask);
   144→  const toggleTask = useFocusly((s) => s.toggleTask);
   145→  const removeTask = useFocusly((s) => s.removeTask);
   146→  const clearDoneTasks = useFocusly((s) => s.clearDoneTasks);
   147→  const setActiveTask = useFocusly((s) => s.setActiveTask);
   148→  const setTaskEstimate = useFocusly((s) => s.setTaskEstimate);
   149→  const moveTask = useFocusly((s) => s.moveTask);
   150→  const moveTaskTo = useFocusly((s) => s.moveTaskTo);
   151→  const [value, setValue] = useState("");
   152→  const [estimate, setEstimate] = useState(1);
   153→  const [dueDate, setDueDate] = useState("");
   154→  const [recurrence, setRecurrence] = useState<TaskRecurrence | null>(null);
   155→  const [recDays, setRecDays] = useState(DEFAULT_RECURRENCE_DAYS);
   156→  /* inline edit — only one task at a time */
   157→  const [editingId, setEditingId] = useState<string | null>(null);
   158→  const [editText, setEditText] = useState("");
   159→  const [editDue, setEditDue] = useState("");
   160→  const [editRec, setEditRec] = useState<TaskRecurrence | null>(null);
   161→  const [editDays, setEditDays] = useState(DEFAULT_RECURRENCE_DAYS);
   162→  /* drag-and-drop reorder state (id of the dragged task / hovered target) */
   163→  const [dragId, setDragId] = useState<string | null>(null);
   164→  const [overId, setOverId] = useState<string | null>(null);
   165→  /* view-level filter — never reorders or mutates the store */
   166→  const [filter, setFilter] = useState<TaskFilter>("toutes");
   167→
   168→  /* PWA shortcut « /?tache=nouvelle#outils » — focus the new-task input once, then clean the URL */
   169→  const newTaskInputRef = useRef<HTMLInputElement | null>(null);
   170→  const deepLinkFired = useRef(false);
   171→  useEffect(() => {
   172→    if (new URLSearchParams(window.location.search).get("tache") !== "nouvelle") return;
   173→    const id = window.setTimeout(() => {
   174→      if (deepLinkFired.current) return;
   175→      deepLinkFired.current = true;
   176→      newTaskInputRef.current?.focus();
   177→      window.history.replaceState(null, "", window.location.pathname + window.location.hash);
   178→    }, 400);
   179→    return () => window.clearTimeout(id);
   180→  }, []);
   181→
   182→  const remaining = tasks.filter((t) => !t.done).length;
   183→  const doneCount = tasks.length - remaining;
   184→  const plannedTotal = tasks.reduce((a, t) => a + (t.done ? 0 : t.estimate ?? 1), 0);
   185→  const spentTotal = tasks.reduce((a, t) => a + (t.done ? 0 : t.spent ?? 0), 0);
   186→
   187→  /* « Aujourd’hui » keeps due-today + overdue, « En retard » only overdue — done tasks included */
   188→  const visibleTasks =
   189→    filter === "toutes"
   190→      ? tasks
   191→      : tasks.filter((t) => {
   192→          if (!t.dueDate) return false;
   193→          const d = daysUntil(t.dueDate);
   194→          return filter === "aujourdhui" ? d <= 0 : d < 0;
   195→        });
   196→
   197→  /* Sectioned « Aujourd’hui » view — the three groups partition visibleTasks exactly */
   198→  const overdueTasks = visibleTasks.filter(
   199→    (t) => !t.done && t.dueDate && daysUntil(t.dueDate) < 0,
   200→  );
   201→  const dueTodayTasks = visibleTasks.filter(
   202→    (t) => !t.done && t.dueDate && daysUntil(t.dueDate) === 0,
   203→  );
   204→  const doneInView = visibleTasks.filter((t) => t.done);
   205→
   206→  const clearDrag = () => {
   207→    setDragId(null);
   208→    setOverId(null);
   209→  };
   210→
   211→  const startTaskEdit = (t: TaskItem) => {
   212→    setEditingId(t.id);
   213→    setEditText(t.text);
   214→    setEditDue(t.dueDate ?? "");
   215→    setEditRec(t.recurrence ?? null);
   216→    setEditDays(t.recurrenceDays ?? DEFAULT_RECURRENCE_DAYS);
   217→  };
   218→
   219→  const cancelTaskEdit = () => setEditingId(null);
   220→
   221→  const saveTaskEdit = (e: FormEvent) => {
   222→    e.preventDefault();
   223→    if (!editingId || !editText.trim()) return;
   224→    // dueDate "" clears the due date, editRec null clears the recurrence (store contract);
   225→    // the custom interval is saved only when the recurrence is « custom », cleared otherwise
   226→    updateTask(editingId, {
   227→      text: editText,
   228→      dueDate: editDue,
   229→      recurrence: editRec,
   230→      recurrenceDays: editRec === "custom" ? editDays : null,
   231→    });
   232→    setEditingId(null);
   233→    toast.success("Tâche mise à jour.");
   234→  };
   235→
   236→  /** One-shot sort: due dates first (ascending, YYYY-MM-DD keys sort chronologically), stable otherwise */
   237→  const sortByDueDate = () => {
   238→    if (!tasks.some((t) => t.dueDate)) {
   239→      toast.info("Aucune échéance définie.");
   240→      return;
   241→    }
   242→    useFocusly.setState((s) => {
   243→      const withDue = s.tasks.filter((t) => t.dueDate);
   244→      const without = s.tasks.filter((t) => !t.dueDate);
   245→      withDue.sort((a, b) => {
   246→        const ka = a.dueDate ?? "";
   247→        const kb = b.dueDate ?? "";
   248→        if (ka === kb) return 0;
   249→        return ka < kb ? -1 : 1;
   250→      });
   251→      return { tasks: [...withDue, ...without] };
   252→    });
   253→    toast.success("Tâches triées par échéance.");
   254→  };
   255→
   256→  /** One task row — shared by the flat list and the sectioned « Aujourd’hui » view */
   257→  const renderTask = (t: TaskItem) => {
   258→    const editing = t.id === editingId;
   259→    const due = t.dueDate ?? null;
   260→    const dueDays = due ? daysUntil(due) : null;
   261→    /* chevron bounds follow the store order — the filter is view-level only */
   262→    const storeIndex = tasks.findIndex((x) => x.id === t.id);
   263→    return (
   264→      <li
   265→        key={t.id}
   266→        draggable={!editing}
   267→        onDragStart={(e) => {
   268→          setDragId(t.id);
   269→          e.dataTransfer.effectAllowed = "move";
   270→          try {
   271→            e.dataTransfer.setData("text/plain", t.id);
   272→          } catch {
   273→            /* some engines forbid setData — the local state is enough */
   274→          }
   275→        }}
   276→        onDragEnd={clearDrag}
   277→        onDragOver={(e) => {
   278→          if (t.done) {
   279→            // Never drop onto a completed task (keeps to-do / done zones separate)
   280→            e.dataTransfer.dropEffect = "none";
   281→            return;
   282→          }
   283→          e.preventDefault();
   284→          e.dataTransfer.dropEffect = "move";
   285→          setOverId((v) => (dragId && dragId !== t.id ? t.id : v));
   286→        }}
   287→        onDragLeave={() => setOverId((v) => (v === t.id ? null : v))}
   288→        onDrop={(e) => {
   289→          e.preventDefault();
   290→          if (t.done || !dragId || dragId === t.id) {
   291→            clearDrag();
   292→            return;
   293→          }
   294→          moveTaskTo(dragId, t.id);
   295→          clearDrag();
   296→        }}
   297→        className={`flex flex-wrap items-start gap-x-2 gap-y-2 rounded-xl border px-3 py-3 transition-[opacity,box-shadow,border-color,transform] duration-150 ${
   298→          editing ? "bg-brand/5 ring-1 ring-brand" : "bg-secondary"
   299→        } ${t.done ? "opacity-55" : ""} ${
   300→          !editing && t.active ? "border-brand/60 ring-1 ring-brand/40" : ""
   301→        } ${dragId === t.id ? "scale-[0.99] opacity-40" : ""} ${
   302→          overId === t.id && dragId !== t.id ? "border-brand/70 ring-2 ring-brand/35" : ""
   303→        } ${dragId && t.done ? "cursor-not-allowed" : ""}`}
   304→      >
   305→        {editing ? (
   306→          <form
   307→            className="flex w-full flex-col gap-2"
   308→            autoComplete="off"
   309→            onSubmit={saveTaskEdit}
   310→            onKeyDown={(e) => {
   311→              if (e.key === "Escape") {
   312→                e.preventDefault();
   313→                cancelTaskEdit();
   314→              }
   315→            }}
   316→          >
   317→            <div className="flex flex-wrap items-center gap-2">
   318→              <label htmlFor={`task-edit-text-${t.id}`} className="sr-only">
   319→                Texte de la tâche
   320→              </label>
   321→              <Input
   322→                id={`task-edit-text-${t.id}`}
   323→                value={editText}
   324→                onChange={(e) => setEditText(e.target.value)}
   325→                maxLength={200}
   326→                className="min-h-11 min-w-[140px] flex-1 rounded-xl"
   327→              />
   328→              <Input
   329→                type="date"
   330→                value={editDue}
   331→                onChange={(e) => setEditDue(e.target.value)}
   332→                aria-label="Échéance de la tâche"
   333→                className="h-11 w-[122px] rounded-xl border bg-secondary/60 px-2 text-xs [color-scheme:light] dark:bg-secondary/60 dark:[color-scheme:dark] sm:h-9"
   334→              />
   335→              {editDue && (
   336→                <button
   337→                  type="button"
   338→                  onClick={() => setEditDue("")}
   339→                  aria-label="Retirer l’échéance"
   340→                  title="Retirer l’échéance"
   341→                  className="grid size-8 place-items-center rounded-md text-faint transition-colors hover:bg-accent hover:text-foreground"
   342→                >
   343→                  <X className="size-3.5" />
   344→                </button>
   345→              )}
   346→            </div>
   347→            <RecurrencePicker
   348→              value={editRec}
   349→              onChange={setEditRec}
   350→              label="Répétition de la tâche"
   351→              days={editDays}
   352→              onDaysChange={setEditDays}
   353→            />
   354→            <div className="flex gap-2">
   355→              <Button
   356→                type="submit"
   357→                disabled={!editText.trim()}
   358→                className="min-h-11 flex-1 rounded-xl active:scale-[0.98]"
   359→              >
   360→                <Check className="size-4" /> Enregistrer
   361→              </Button>
   362→              <Button
   363→                type="button"
   364→                variant="outline"
   365→                onClick={cancelTaskEdit}
   366→                className="min-h-11 rounded-xl"
   367→              >
   368→                <X className="size-4" /> Annuler
   369→              </Button>
   370→            </div>
   371→          </form>
   372→        ) : (
   373→          <>
   374→            <GripVertical
   375→              aria-hidden
   376→              className="mt-0.5 size-4 shrink-0 cursor-grab text-faint/60 transition-colors hover:text-muted-foreground active:cursor-grabbing"
   377→            />

   379→              checked={t.done}
   380→              onCheckedChange={(v) => {
   381→                const nextDone = v === true;
   382→                if (nextDone && !t.done && t.recurrence) {
   383→                  toggleTask(t.id, true);

   385→                    `Prochaine occurrence planifiée : ${frDateShort(
   386→                      nextDueDate(t.dueDate, t.recurrence, t.recurrenceDays),
   387→                    )}`,
   388→                  );

   390→                  toggleTask(t.id, nextDone);
   391→                }
   392→              }}
   393→              aria-label={t.done ? "Marquer comme à faire" : "Marquer comme terminée"}
   394→              className="mt-0.5 size-5"
   395→            />
   396→            <span
   397→              className={`min-w-[140px] flex-1 basis-[140px] break-words text-sm leading-relaxed ${
   398→                t.done ? "line-through" : ""
   399→              }`}

   408→                    ? "border-destructive/30 bg-destructive/15 text-destructive"
   409→                    : dueDays === 0

   415→                  <CalendarClock className="size-[11px]" />

   417→                  <Calendar className="size-[11px]" />
   418→                )}
   419→                {frDateShort(due)}
   420→              </span>
   421→            )}
   422→            {t.recurrence && (
   423→              <span
   424→                title={
   425→                  t.recurrence === "custom"
   426→                    ? `Répétition tous les ${t.recurrenceDays ?? DEFAULT_RECURRENCE_DAYS} jours`
   427→                    : `Répétition ${RECURRENCE_LABELS[t.recurrence].toLowerCase()}`
   428→                }
   429→                className="inline-flex shrink-0 items-center gap-1 rounded-full border bg-secondary px-2 py-0.5 text-[11px] text-muted-foreground"

   431→                <Repeat className="size-3" />

   441→                className="flex shrink-0 items-center gap-0.5 rounded-lg border bg-background/60"

   443→                <button
   444→                  type="button"
   445→                  aria-label="Diminuer l’estimation"
   446→                  onClick={() => setTaskEstimate(t.id, (t.estimate ?? 1) - 1)}
   447→                  disabled={(t.estimate ?? 1) <= 1}
   448→                  className="grid size-6 place-items-center rounded-l-md text-faint transition-colors hover:bg-accent hover:text-foreground disabled:opacity-30 disabled:hover:bg-transparent"
   449→                >
   450→                  <Minus className="size-3" />
   451→                </button>
   452→                <span
   453→                  title={`Estimation : ${t.estimate ?? 1} pomodoro${(t.estimate ?? 1) > 1 ? "s" : ""} · Effectués : ${t.spent ?? 0}`}
   454→                  className={`w-9 text-center text-[11px] font-semibold tabular-nums ${
   455→                    (t.spent ?? 0) >= (t.estimate ?? 1) ? "text-brand" : "text-muted-foreground"
   456→                  }`}
   457→                >
   458→                  {t.spent ?? 0}/{t.estimate ?? 1}
   459→                </span>
   460→                <button
   461→                  type="button"
   462→                  aria-label="Augmenter l’estimation"
   463→                  onClick={() => setTaskEstimate(t.id, (t.estimate ?? 1) + 1)}
   464→                  disabled={(t.estimate ?? 1) >= 12}
   465→                  className="grid size-6 place-items-center rounded-r-md text-faint transition-colors hover:bg-accent hover:text-foreground disabled:opacity-30 disabled:hover:bg-transparent"
   466→                >
   467→                  <Plus className="size-3" />
   468→                </button>
   469→              </div>
   470→            )}
   471→            {!t.done && (
   472→              <button
   473→                onClick={() => setActiveTask(t.id)}
   474→                aria-label="Lier au minuteur"
   475→                title="Lier au minuteur"
   476→                className={`transition-colors ${
   477→                  t.active ? "text-brand" : "text-faint hover:text-foreground"
   478→                }`}
   479→              >
   480→                <Crosshair className="size-4" />
   481→              </button>

   483→            <button
   484→              type="button"
   485→              onClick={() => startTaskEdit(t)}
   486→              aria-label={`Modifier la tâche : ${t.text}`}
   487→              title="Modifier la tâche"
   488→              className="text-faint transition-colors hover:text-foreground"
   489→            >
   490→              <Pencil className="size-4" />
   491→            </button>

   493→              onClick={() => removeTask(t.id)}

   495→              className="text-faint transition-colors hover:text-destructive"
   496→            >
   497→              <Trash2 className="size-4" />
   498→            </button>
   499→            {tasks.length > 1 && (
   500→              <div
   501→                role="group"
   502→                aria-label={`Déplacer la tâche : ${t.text}`}
   503→                className="flex shrink-0 flex-col gap-0.5"
   504→              >
   505→                <button
   506→                  type="button"
   507→                  onClick={() => moveTask(t.id, -1)}
   508→                  disabled={storeIndex === 0}
   509→                  aria-label={`Monter la tâche : ${t.text}`}
   510→                  className="grid size-4 place-items-center text-faint transition-colors hover:text-foreground disabled:opacity-25 disabled:hover:text-faint"
   511→                >
   512→                  <ChevronUp className="size-3.5" />
   513→                </button>
   514→                <button
   515→                  type="button"
   516→                  onClick={() => moveTask(t.id, 1)}
   517→                  disabled={storeIndex === tasks.length - 1}
   518→                  aria-label={`Descendre la tâche : ${t.text}`}
   519→                  className="grid size-4 place-items-center text-faint transition-colors hover:text-foreground disabled:opacity-25 disabled:hover:text-faint"
   520→                >
   521→                  <ChevronDown className="size-3.5" />
   522→                </button>
   523→              </div>
   524→            )}
   525→            {t.done && t.recurrence && (
   526→              <p className="w-full text-xs text-faint">
   527→                Prochaine occurrence : {frDateShort(nextDueDate(t.dueDate, t.recurrence, t.recurrenceDays))}
   528→              </p>
   529→            )}
   530→          </>
   531→        )}
   532→      </li>
   533→    );
   534→  };
   535→
   536→  return (
   537→    <section
   538→      aria-label="Tâches"
   539→      className="flex flex-col gap-4 rounded-3xl border bg-card p-6"
   540→    >
   541→      <div className="flex flex-wrap items-center justify-between gap-2">
   542→        <h2 className="m-0 text-base font-bold tracking-tight">Tâches</h2>
   543→        <div className="flex items-center gap-1.5">
   544→          {tasks.length >= 2 && (
   545→            <Button
   546→              variant="ghost"
   547→              size="sm"
   548→              onClick={sortByDueDate}
   549→              className="text-muted-foreground"
   550→            >
   551→              <ArrowDownWideNarrow className="size-4" />
   552→              Trier par échéance
   553→            </Button>
   554→          )}
   555→          <span className="text-xs font-semibold text-muted-foreground">
   556→            {remaining} / {tasks.length}
   557→          </span>
   558→          {filter !== "toutes" && (
   559→            <span className="text-xs text-faint">
   560→              · {visibleTasks.length} affichée{visibleTasks.length > 1 ? "s" : ""}
   561→            </span>
   562→          )}
   563→        </div>
   564→      </div>
   565→
   566→      <div
   567→        role="group"
   568→        aria-label="Filtrer les tâches"
   569→        className="-mt-2 flex flex-wrap items-center gap-1.5"
   570→      >
   571→        {TASK_FILTERS.map((f) => (
   572→          <button
   573→            key={f.value}
   574→            type="button"
   575→            aria-pressed={filter === f.value}
   576→            onClick={() => setFilter(f.value)}
   577→            className={`rounded-full border px-2.5 py-1 text-xs transition-colors ${
   578→              filter === f.value
   579→                ? "border-brand/40 bg-brand/15 font-medium text-brand"
   580→                : "text-muted-foreground hover:text-foreground"
   581→            }`}
   582→          >
   583→            {f.label}
   584→          </button>
   585→        ))}
   586→      </div>
   587→
   588→      {tasks.length > 0 && remaining > 0 && (
   589→        <p className="-mt-2 text-[11.5px] text-faint" aria-live="polite">
   590→          Progression : {spentTotal} sur {plannedTotal} pomodoro{plannedTotal > 1 ? "s" : ""}{" "}
   591→          estimé{plannedTotal > 1 ? "s" : ""} — tâches en cours
   592→        </p>
   593→      )}
   594→
   595→      <form
   596→        className="flex flex-wrap gap-2"
   597→        autoComplete="off"
   598→        onSubmit={(e) => {
   599→          e.preventDefault();
   600→          if (!value.trim()) return;
   601→          addTask(
   602→            value,
   603→            estimate,
   604→            dueDate,
   605→            recurrence ?? undefined,
   606→            recurrence === "custom" ? recDays : undefined,
   607→          );
   608→          setValue("");
   609→          setEstimate(1);
   610→          setDueDate("");
   611→          setRecurrence(null);
   612→          setRecDays(DEFAULT_RECURRENCE_DAYS);
   613→        }}
   614→      >
   615→        <label htmlFor="task-input" className="sr-only">
   616→          Nouvelle tâche
   617→        </label>
   618→        <Input
   619→          ref={newTaskInputRef}
   620→          id="task-input"
   621→          value={value}
   622→          onChange={(e) => setValue(e.target.value)}
   623→          placeholder="Ajouter une tâche…"
   624→          maxLength={200}
   625→          className="min-h-11 min-w-[200px] flex-1 rounded-xl"
   626→        />
   627→        <div
   628→          role="group"
   629→          aria-label="Estimation en pomodoros"
   630→          className="flex items-center rounded-xl border bg-secondary"
   631→        >
   632→          <button
   633→            type="button"
   634→            aria-label="Diminuer l’estimation"
   635→            onClick={() => setEstimate((v) => Math.max(1, v - 1))}
   636→            disabled={estimate <= 1}
   637→            className="grid size-9 place-items-center rounded-l-xl text-muted-foreground transition-colors hover:bg-accent hover:text-foreground disabled:opacity-35 disabled:hover:bg-transparent"
   638→          >
   639→            <Minus className="size-3.5" />
   640→          </button>
   641→          <span
   642→            aria-live="polite"
   643→            title={`${estimate} pomodoro${estimate > 1 ? "s" : ""} estimé${estimate > 1 ? "s" : ""}`}
   644→            className="w-9 text-center text-[13px] font-semibold tabular-nums"
   645→          >
   646→            {estimate}
   647→          </span>
   648→          <button
   649→            type="button"
   650→            aria-label="Augmenter l’estimation"
   651→            onClick={() => setEstimate((v) => Math.min(12, v + 1))}
   652→            disabled={estimate >= 12}
   653→            className="grid size-9 place-items-center rounded-r-xl text-muted-foreground transition-colors hover:bg-accent hover:text-foreground disabled:opacity-35 disabled:hover:bg-transparent"
   654→          >
   655→            <Plus className="size-3.5" />
   656→          </button>
   657→        </div>
   658→        <Input
   659→          type="date"
   660→          value={dueDate}
   661→          onChange={(e) => setDueDate(e.target.value)}
   662→          aria-label="Échéance (optionnelle)"
   663→          className="h-9 w-[122px] rounded-xl border bg-secondary/60 px-2 text-xs [color-scheme:light] dark:bg-secondary/60 dark:[color-scheme:dark]"
   664→        />
   665→        <RecurrencePicker
   666→          value={recurrence}
   667→          onChange={setRecurrence}
   668→          label="Répétition de la nouvelle tâche"
   669→          days={recDays}
   670→          onDaysChange={setRecDays}
   671→        />
   672→        <Button type="submit" className="min-h-11 rounded-xl active:scale-[0.98]">
   673→          <Plus className="size-4" /> Ajouter
   674→        </Button>
   675→      </form>
   676→
   677→      {filter === "aujourdhui" ? (
   678→        /* Sectioned « Aujourd’hui » — overdue first, then today's due dates, then done.
   679→           Empty sections are never rendered (no header without content). */
   680→        <div className="slim-scroll flex max-h-96 flex-col gap-4 overflow-y-auto">
   681→          {overdueTasks.length > 0 && (
   682→            <section aria-labelledby="tasks-overdue-heading" className="flex flex-col gap-2">
   683→              <h3
   684→                id="tasks-overdue-heading"
   685→                className="m-0 flex items-center gap-1.5 text-[11px] font-semibold uppercase tracking-wide text-faint"
   686→              >
   687→                <span aria-hidden className="size-1.5 rounded-full bg-destructive/70" />
   688→                En retard · {overdueTasks.length}
   689→              </h3>
   690→              <ul className="m-0 flex list-none flex-col gap-2 p-0">
   691→                {overdueTasks.map(renderTask)}
   692→              </ul>
   693→            </section>
   694→          )}
   695→          {dueTodayTasks.length > 0 && (
   696→            <section aria-labelledby="tasks-today-heading" className="flex flex-col gap-2">
   697→              <h3
   698→                id="tasks-today-heading"
   699→                className="m-0 text-[11px] font-semibold uppercase tracking-wide text-faint"
   700→              >
   701→                Échéances du jour · {dueTodayTasks.length}
   702→              </h3>
   703→              <ul className="m-0 flex list-none flex-col gap-2 p-0">
   704→                {dueTodayTasks.map(renderTask)}
   705→              </ul>
   706→            </section>
   707→          )}
   708→          {doneInView.length > 0 && (
   709→            <section aria-labelledby="tasks-done-heading" className="flex flex-col gap-2">
   710→              <h3
   711→                id="tasks-done-heading"
   712→                className="m-0 text-[11px] font-semibold uppercase tracking-wide text-faint"
   713→              >
   714→                Terminées · {doneInView.length}
   715→              </h3>
   716→              <ul className="m-0 flex list-none flex-col gap-2 p-0">
   717→                {doneInView.map(renderTask)}
   718→              </ul>
   719→            </section>
   720→          )}
   721→        </div>
   722→      ) : (
   723→        <ul className="slim-scroll flex max-h-96 list-none flex-col gap-2 overflow-y-auto p-0 m-0">
   724→          {visibleTasks.map(renderTask)}
   725→        </ul>
   726→      )}
   727→
   728→      {tasks.length === 0 && (
   729→        <div className="rounded-xl border border-dashed px-4 py-6 text-center text-[13.5px] text-faint">
   730→          Aucune tâche pour le moment. Ajoutez-en une pour la lier au minuteur.
   731→        </div>
   732→      )}
   733→
   734→      {tasks.length > 0 && visibleTasks.length === 0 && filter !== "toutes" && (
   735→        filter === "retard" ? (
   736→          <div className="flex flex-col items-center gap-1.5 rounded-xl border border-dashed px-4 py-6 text-center text-[13.5px] text-faint">
   737→            <CheckCircle2 aria-hidden className="size-5 text-brand" />
   738→            Aucune tâche en retard. Vous êtes à jour.
   739→          </div>
   740→        ) : (
   741→          <div className="rounded-xl border border-dashed px-4 py-6 text-center text-[13.5px] text-faint">
   742→            Rien de prévu aujourd’hui. Profitez-en ou planifiez une échéance.
   743→          </div>
   744→        )
   745→      )}
   746→
   747→      {doneCount > 0 && (
   748→        <Button variant="ghost" size="sm" className="self-start" onClick={clearDoneTasks}>
   749→          Effacer les tâches terminées ({doneCount})
   750→        </Button>
   751→      )}
   752→    </section>
   753→  );
   754→}
