    76→function levelIndex(pomodoros: number): number {
    77→  if (pomodoros <= 0) return 0;
    78→  if (pomodoros <= 2) return 1;
    79→  if (pomodoros <= 4) return 2;
    80→  if (pomodoros <= 6) return 3;
    81→  return 4;
    82→}

    84→/** Local-date key — same decomposition as todayKey() in types.ts. */
    85→function dayKeyOf(y: number, m0: number, d: number): string {
    86→  return `${y}-${String(m0 + 1).padStart(2, "0")}-${String(d).padStart(2, "0")}`;
    87→}

    89→/** Number of days in a month (leap years handled by the Date constructor). */
    90→function daysInMonthOf(y: number, m0: number): number {
    91→  return new Date(y, m0 + 1, 0).getDate();
    92→}
    93→
    94→/** Sum of pomodoros over every day of a month — the same derivation as the
    95→ * monthly calendar (local dayKeyOf keys, EMPTY_STAT fallback); shared by the
    96→ * calendar and the monthly goal gauge so the math lives in one place. */
    97→function monthPomodorosOf(daily: Record<string, DailyStat>, y: number, m0: number): number {
    98→  const days = daysInMonthOf(y, m0);
    99→  let sum = 0;
   100→  for (let d = 1; d <= days; d++) {
   101→    sum += (daily[dayKeyOf(y, m0, d)] ?? EMPTY_STAT).pomodoros;
   102→  }
   103→  return sum;
   104→}
   105→
   106→/** Number of days in a year (sum of month lengths — leap years included). */
   107→function daysInYearOf(y: number): number {
   108→  let days = 0;
   109→  for (let m = 0; m < 12; m++) days += daysInMonthOf(y, m);
   110→  return days;
   111→}
   112→
   113→/** Sum of pomodoros over every day of a year — same derivation as the monthly
   114→ * calendar (local dayKeyOf keys via the monthly sums, EMPTY_STAT fallback);
   115→ * shared by the year heatmap zone and the annual goal gauge. */
   116→function yearPomodorosOf(daily: Record<string, DailyStat>, y: number): number {
   117→  let sum = 0;
   118→  for (let m = 0; m < 12; m++) sum += monthPomodorosOf(daily, y, m);
   119→  return sum;
   120→}

   176→    let seconds = 0;
   177→    let activeDays = 0;

   179→    Object.values(daily).forEach((s) => {
   180→      pomodoros += s.pomodoros;
   181→      seconds += s.focusSeconds;
   182→      if (s.pomodoros > 0) activeDays += 1;

   185→    const minutes = Math.floor((seconds % 3600) / 60);

   187→      pomodoros,
   188→      activeDays,
   189→      recorded,

   212→        recorded += 1;
   213→        pomodoros += s.pomodoros;
   214→        seconds += s.focusSeconds;
   215→        if (s.pomodoros > 0) activeDays += 1;

   219→    const minutes = Math.floor((seconds % 3600) / 60);

   222→      activeDays,
   223→      recorded,

   250→          pomodoros += s.pomodoros;
   251→          seconds += s.focusSeconds;
   252→        }

   341→      perDay,
   342→      daysInYear,

   348→      onPace: done >= perDay * dayOfYear,

   467→            firstDay.toLocaleDateString("fr-FR", { month: "short" }).replace(".", ""),

   469→          week,
   470→        });

   472→      }

   490→
   491→  const now = new Date();
   492→  const isCurrentMonth = view.y === now.getFullYear() && view.m === now.getMonth();
   493→  const isCurrentYear = viewYear === now.getFullYear();
   494→
   495→  const goPrevMonth = () =>
   496→    setView((v) => (v.m === 0 ? { y: v.y - 1, m: 11 } : { ...v, m: v.m - 1 }));

   498→    setView((v) => (v.m === 11 ? { y: v.y + 1, m: 0 } : { ...v, m: v.m + 1 }));
   499→
   500→  const gridAria = `${monthLabel} : ${month.monthPomodoros} ${
   501→    month.monthPomodoros === 1 ? "pomodoro" : "pomodoros"
   502→  } sur ${totals.pomodoros} ${totals.pomodoros === 1 ? "pomodoro" : "pomodoros"} enregistré${
   503→    totals.pomodoros === 1 ? "" : "s"
   504→  }`;
   505→
   506→  const yearAria = `Année ${viewYear} : ${yearData.yearPomodoros} ${
   507→    yearData.yearPomodoros === 1 ? "pomodoro" : "pomodoros"
   508→  } sur ${totals.pomodoros} ${totals.pomodoros === 1 ? "pomodoro" : "pomodoros"} enregistré${
   509→    totals.pomodoros === 1 ? "" : "s"
   510→  }`;
   511→
   512→  const hasData = totals.recorded > 0;
   513→
   514→  /* ----- Comparaison hebdomadaire : libellés dérivés ----- */
   515→  const weekMax = Math.max(weekCompare.current.pomodoros, weekCompare.previous.pomodoros);
   516→  const weekAria = `Cette semaine : ${weekCompare.current.pomodoros} ${
   517→    weekCompare.current.pomodoros === 1 ? "pomodoro" : "pomodoros"
   518→  }. Semaine dernière : ${weekCompare.previous.pomodoros} ${
   519→    weekCompare.previous.pomodoros === 1 ? "pomodoro" : "pomodoros"
