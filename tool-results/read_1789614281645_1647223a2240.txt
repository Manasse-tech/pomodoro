    71→function levelIndex(pomodoros: number): number {
    72→  if (pomodoros <= 0) return 0;
    73→  if (pomodoros <= 2) return 1;
    74→  if (pomodoros <= 4) return 2;
    75→  if (pomodoros <= 6) return 3;
    76→  return 4;
    77→}

    79→/** Local-date key — same decomposition as todayKey() in types.ts. */
    80→function dayKeyOf(y: number, m0: number, d: number): string {
    81→  return `${y}-${String(m0 + 1).padStart(2, "0")}-${String(d).padStart(2, "0")}`;
    82→}
    83→
    84→/** Number of days in a month (leap years handled by the Date constructor). */
    85→function daysInMonthOf(y: number, m0: number): number {
    86→  return new Date(y, m0 + 1, 0).getDate();
    87→}
    88→
    89→/** Sum of pomodoros over every day of a month — the same derivation as the
    90→ * monthly calendar (local dayKeyOf keys, EMPTY_STAT fallback); shared by the
    91→ * calendar and the monthly goal gauge so the math lives in one place. */
    92→function monthPomodorosOf(daily: Record<string, DailyStat>, y: number, m0: number): number {
    93→  const days = daysInMonthOf(y, m0);
    94→  let sum = 0;
    95→  for (let d = 1; d <= days; d++) {
    96→    sum += (daily[dayKeyOf(y, m0, d)] ?? EMPTY_STAT).pomodoros;
    97→  }
    98→  return sum;
    99→}

   482→  /* ----- KPI cards (scoped to the selected period) ----- */
   483→  const shown = periodTotals ?? totals;

   485→    period === null

   488→  // Libellé condensé pour l’image partagée (pas de phrase complète)

   490→
   491→  const kpis: Array<{
