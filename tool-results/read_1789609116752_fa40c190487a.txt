   119→   119→export function StatistiquesView() {

   122→   122→  const [view, setView] = useState<ViewMonth>(() => {

   132→   132→  /* ----- Grand totals across all recorded days ----- */

   135→   135→    let seconds = 0;
   136→   136→    let activeDays = 0;
   137→   137→    const recorded = Object.keys(daily).length;

   139→   139→      pomodoros += s.pomodoros;
   140→   140→      seconds += s.focusSeconds;
   141→   141→      if (s.pomodoros > 0) activeDays += 1;

   144→   144→    const minutes = Math.floor((seconds % 3600) / 60);

   146→   146→      pomodoros,
   147→   147→      activeDays,
   148→   148→      recorded,
   149→   149→      focusSeconds: seconds,
   150→   150→      focusLabel:
   151→   151→        hours > 0 ? `${hours} h ${String(minutes).padStart(2, "0")} min` : `${minutes} min`,
   152→   152→    };
   153→   153→  }, [daily]);
   154→   154→
   155→   155→  /* ----- Totals within the selected period (null when no filter) ----- */

   160→   160→    const cutoff = new Date(n.getFullYear(), n.getMonth(), n.getDate());
   161→   161→    cutoff.setDate(cutoff.getDate() - (period - 1));

   163→   163→    let pomodoros = 0;

   166→   166→    let recorded = 0;

   168→   168→      // Zero-padded keys → lexicographic comparison is date comparison

   171→   171→        recorded += 1;
   172→   172→        pomodoros += s.pomodoros;
   173→   173→        seconds += s.focusSeconds;
   174→   174→        if (s.pomodoros > 0) activeDays += 1;
   175→   175→      }

   181→   181→      activeDays,
   182→   182→      recorded,
   183→   183→      focusSeconds: seconds,
   184→   184→      focusLabel:

   203→   203→      let pomodoros = 0;
   204→   204→      let seconds = 0;
   205→   205→      const cur = new Date(start);
   206→   206→      for (let i = 0; i < 7; i++) {
   207→   207→        const s = daily[dayKeyOf(cur.getFullYear(), cur.getMonth(), cur.getDate())];
   208→   208→        if (s) {
   209→   209→          pomodoros += s.pomodoros;
   210→   210→          seconds += s.focusSeconds;
   211→   211→        }
   212→   212→        cur.setDate(cur.getDate() + 1);
   213→   213→      }
   214→   214→      const hours = Math.floor(seconds / 3600);
   215→   215→      const minutes = Math.floor((seconds % 3600) / 60);

   217→   217→        pomodoros,
   218→   218→        focusLabel:
   219→   219→          hours > 0 ? `${hours} h ${String(minutes).padStart(2, "0")} min` : `${minutes} min`,
   220→   220→      };
   221→   221→    };
   222→   222→
   223→   223→    const current = sumWeek(monday);
   224→   224→    const previous = sumWeek(prevMonday);

   226→   226→    // Position dans la semaine : 0 = lundi … 6 = dimanche
   227→   227→    const dayIndex = (todayMid.getDay() + 6) % 7;
   228→   228→    const daysElapsed = dayIndex + 1;
   229→   229→
   230→   230→    // Évolution en % — garde anti-division par zéro

   232→   232→    let pct = 0;
   233→   233→    if (previous.pomodoros === 0 && current.pomodoros > 0) {
   234→   234→      trend = "new";
   235→   235→    } else if (previous.pomodoros > 0) {
   236→   236→      pct = Math.round(((current.pomodoros - previous.pomodoros) / previous.pomodoros) * 100);
   237→   237→      trend = pct > 0 ? "up" : pct < 0 ? "down" : "flat";
   238→   238→    }
   239→   239→
   240→   240→    return { current, previous, daysElapsed, weekComplete: dayIndex === 6, trend, pct };
   241→   241→  }, [daily]);

   243→   243→  /* ----- Current month grid ----- */

   247→   247→        month: "long",
   248→   248→        year: "numeric",
   249→   249→      }),

   251→   251→  }, [view]);
   252→   252→
   253→   253→  const month = useMemo(() => {
   254→   254→    const now = new Date();
   255→   255→    const todayMid = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   256→   256→    const daysInMonth = new Date(view.y, view.m + 1, 0).getDate();
   257→   257→    // French weeks start Monday: JS Sunday=0 → Monday-first index
   258→   258→    const leading = (new Date(view.y, view.m, 1).getDay() + 6) % 7;
   259→   259→
   260→   260→    const cells: MonthCell[] = [];
   261→   261→    for (let d = 1; d <= daysInMonth; d++) {
   262→   262→      const key = dayKeyOf(view.y, view.m, d);
   263→   263→      const stat = daily[key] ?? EMPTY_STAT;
   264→   264→      const pomodoros = stat.pomodoros;
   265→   265→      const minutes = Math.floor(stat.focusSeconds / 60);
   266→   266→      const dateStr = new Date(view.y, view.m, d).toLocaleDateString("fr-FR", {
   267→   267→        day: "numeric",
   268→   268→        month: "long",
   269→   269→        year: "numeric",
   270→   270→      });
   271→   271→      cells.push({
   272→   272→        key,
   273→   273→        pomodoros,

   275→   275→        isToday: key === todayKey(),
   276→   276→        title:
   277→   277→          pomodoros === 0
   278→   278→            ? `${dateStr} — aucun pomodoro`
   279→   279→            : `${dateStr} — ${pomodoros} pomodoro${pomodoros > 1 ? "s" : ""}, ${minutes} min de concentration`,
   280→   280→      });
   281→   281→    }
   282→   282→
   283→   283→    const monthPomodoros = cells.reduce((a, c) => a + c.pomodoros, 0);
   284→   284→    const trailing = (7 - ((leading + daysInMonth) % 7)) % 7;
   285→   285→    return { leading, trailing, cells, monthPomodoros };
   286→   286→  }, [daily, view]);
   287→   287→
   288→   288→  /* ----- Year heatmap (53 weeks × 7 days, Monday-first) ----- */
   289→   289→  const yearData = useMemo(() => {

   291→   291→    const todayMid = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   292→   292→    const first = new Date(viewYear, 0, 1);
   293→   293→    // Start on the Monday on or before January 1st

   295→   295→    start.setDate(first.getDate() - ((first.getDay() + 6) % 7));
   296→   296→    const end = new Date(viewYear, 11, 31);

   298→   298→    const weeks: YearDayCell[][] = [];

   307→   307→        const key = dayKeyOf(cur.getFullYear(), cur.getMonth(), cur.getDate());

   311→   311→        if (inYear && !isFuture && pomodoros > 0) {
   312→   312→          yearPomodoros += pomodoros;
   313→   313→          yearActiveDays += 1;
   314→   314→        }
   315→   315→        const minutes = Math.floor((daily[key] ?? EMPTY_STAT).focusSeconds / 60);
   316→   316→        const dateStr = new Date(cur).toLocaleDateString("fr-FR", {
   317→   317→          day: "numeric",
   318→   318→          month: "long",
   319→   319→          year: "numeric",
   320→   320→        });

   322→   322→          key,
   323→   323→          pomodoros,
   324→   324→          inYear,
   325→   325→          isFuture,
   326→   326→          isToday,
   327→   327→          title:
   328→   328→            !inYear || isFuture
   329→   329→              ? ""
   330→   330→              : pomodoros === 0
   331→   331→                ? `${dateStr} — aucun pomodoro`
   332→   332→                : `${dateStr} — ${pomodoros} pomodoro${pomodoros > 1 ? "s" : ""}, ${minutes} min de concentration`,
   333→   333→        });
   334→   334→        cur.setDate(cur.getDate() + 1);
   335→   335→      }
   336→   336→      weeks.push(week);
   337→   337→    }
   338→   338→
   339→   339→    // Month marks: week index of the 1st of each month, thinned to avoid overlap
   340→   340→    const marks: MonthMark[] = [];

   345→   345→      const week = Math.floor(diff / 7);
   346→   346→      if (week - lastWeek >= 3) {
   347→   347→        marks.push({
   348→   348→          label: capitalize(
   349→   349→            firstDay.toLocaleDateString("fr-FR", { month: "short" }).replace(".", ""),
   350→   350→          ),
   351→   351→          week,
   352→   352→        });
   353→   353→        lastWeek = week;
   354→   354→      }
   355→   355→    }
   356→   356→
   357→   357→    return { weeks, yearPomodoros, yearActiveDays, marks };
   358→   358→  }, [daily, viewYear]);
   359→   359→

   367→   367→    });
   368→   368→    const max = Math.max(...totals);
   369→   369→    const bestIndex = totals.indexOf(max);
   370→   370→    return { totals, max, bestIndex };
   371→   371→  }, [daily]);
   372→   372→
   373→   373→  const now = new Date();
   374→   374→  const isCurrentMonth = view.y === now.getFullYear() && view.m === now.getMonth();
   375→   375→  const isCurrentYear = viewYear === now.getFullYear();
   376→   376→
   377→   377→  const goPrevMonth = () =>
   378→   378→    setView((v) => (v.m === 0 ? { y: v.y - 1, m: 11 } : { ...v, m: v.m - 1 }));
   379→   379→  const goNextMonth = () =>
   380→   380→    setView((v) => (v.m === 11 ? { y: v.y + 1, m: 0 } : { ...v, m: v.m + 1 }));
   381→   381→
   382→   382→  const gridAria = `${monthLabel} : ${month.monthPomodoros} ${
   383→   383→    month.monthPomodoros === 1 ? "pomodoro" : "pomodoros"
   384→   384→  } sur ${totals.pomodoros} ${totals.pomodoros === 1 ? "pomodoro" : "pomodoros"} enregistré${
   385→   385→    totals.pomodoros === 1 ? "" : "s"
   386→   386→  }`;
   387→   387→
   388→   388→  const yearAria = `Année ${viewYear} : ${yearData.yearPomodoros} ${
   389→   389→    yearData.yearPomodoros === 1 ? "pomodoro" : "pomodoros"
   390→   390→  } sur ${totals.pomodoros} ${totals.pomodoros === 1 ? "pomodoro" : "pomodoros"} enregistré${
   391→   391→    totals.pomodoros === 1 ? "" : "s"
   392→   392→  }`;
   393→   393→
   394→   394→  const hasData = totals.recorded > 0;
   395→   395→
   396→   396→  /* ----- Comparaison hebdomadaire : libellés dérivés ----- */
   397→   397→  const weekMax = Math.max(weekCompare.current.pomodoros, weekCompare.previous.pomodoros);
   398→   398→  const weekAria = `Cette semaine : ${weekCompare.current.pomodoros} ${
   399→   399→    weekCompare.current.pomodoros === 1 ? "pomodoro" : "pomodoros"
   400→   400→  }. Semaine dernière : ${weekCompare.previous.pomodoros} ${
   401→   401→    weekCompare.previous.pomodoros === 1 ? "pomodoro" : "pomodoros"
   402→   402→  }.`;
   403→   403→  // Largeur de barre (max des deux semaines = pleine largeur, plancher 2 % si actif)
   404→   404→  const weekBarPct = (v: number) =>
   405→   405→    weekMax > 0 ? Math.max((v / weekMax) * 100, 2) : 0;
   406→   406→
   407→   407→  const handleExport = () => {
   408→   408→    if (Object.keys(daily).length === 0) {
   409→   409→      toast.info("Aucune donnée à exporter.");
   410→   410→      return;
   411→   411→    }
   412→   412→    downloadTextFile(dailyToCsv(daily), `focusly-statistiques-${todayKey()}.csv`);
   413→   413→    toast.success("Export CSV téléchargé.");
   414→   414→  };
   415→   415→
   416→   416→  /* ----- KPI cards (scoped to the selected period) ----- */
   417→   417→  const shown = periodTotals ?? totals;
   418→   418→  const periodLabel =
   419→   419→    period === null
   420→   420→      ? "Sur l’ensemble de votre historique."
   421→   421→      : `${period} derniers jours, aujourd’hui inclus.`;
   422→   422→  // Libellé condensé pour l’image partagée (pas de phrase complète)
   423→   423→  const imagePeriodLabel = period === null ? "Historique complet" : `${period} derniers jours`;
   424→   424→
   425→   425→  const kpis: Array<{
   426→   426→    icon: LucideIcon;
   427→   427→    value: string;
   428→   428→    label: string;
   429→   429→    hint?: string;
   430→   430→    hint2?: string;
   431→   431→  }> = [
   432→   432→    {
   433→   433→      icon: Flame,
   434→   434→      value: String(shown.pomodoros),
   435→   435→      label: "Pomodoros",
   436→   436→    },
   437→   437→    {
   438→   438→      icon: Timer,
   439→   439→      value: shown.focusLabel,
   440→   440→      label: "Temps de concentration",
