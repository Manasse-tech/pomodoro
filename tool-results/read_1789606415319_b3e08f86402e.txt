   113→   113→export function StatistiquesView() {
   114→   114→  const daily = useFocusly((s) => s.daily);
   115→   115→  const [calendarTab, setCalendarTab] = useState<"mensuel" | "annuel">("mensuel");
   116→   116→  const [view, setView] = useState<ViewMonth>(() => {
   117→   117→    const n = new Date();
   118→   118→    return { y: n.getFullYear(), m: n.getMonth() };
   119→   119→  });
   120→   120→  const [viewYear, setViewYear] = useState<number>(() => new Date().getFullYear());
   121→   121→  /** Selected KPI period in days — null = all time (default, preserves old behavior). */
   122→   122→  const [period, setPeriod] = useState<7 | 30 | 90 | null>(null);
   123→   123→
   124→   124→  /* ----- Grand totals across all recorded days ----- */
   125→   125→  const totals = useMemo(() => {
   126→   126→    let pomodoros = 0;
   127→   127→    let seconds = 0;
   128→   128→    let activeDays = 0;
   129→   129→    const recorded = Object.keys(daily).length;
   130→   130→    Object.values(daily).forEach((s) => {
   131→   131→      pomodoros += s.pomodoros;
   132→   132→      seconds += s.focusSeconds;
   133→   133→      if (s.pomodoros > 0) activeDays += 1;
   134→   134→    });
   135→   135→    const hours = Math.floor(seconds / 3600);
   136→   136→    const minutes = Math.floor((seconds % 3600) / 60);
   137→   137→    return {
   138→   138→      pomodoros,
   139→   139→      activeDays,
   140→   140→      recorded,
   141→   141→      focusLabel:
   142→   142→        hours > 0 ? `${hours} h ${String(minutes).padStart(2, "0")} min` : `${minutes} min`,
   143→   143→    };
   144→   144→  }, [daily]);
   145→   145→
   146→   146→  /* ----- Totals within the selected period (null when no filter) ----- */
   147→   147→  const periodTotals = useMemo(() => {
   148→   148→    if (period === null) return null;
   149→   149→    const n = new Date();
   150→   150→    // Cutoff = today minus (N-1) days → today is always included
   151→   151→    const cutoff = new Date(n.getFullYear(), n.getMonth(), n.getDate());
   152→   152→    cutoff.setDate(cutoff.getDate() - (period - 1));
   153→   153→    const cutoffKey = todayKey(cutoff); // zero-padded YYYY-MM-DD (local)
   154→   154→    let pomodoros = 0;
   155→   155→    let seconds = 0;
   156→   156→    let activeDays = 0;
   157→   157→    let recorded = 0;
   158→   158→    Object.keys(daily).forEach((k) => {
   159→   159→      // Zero-padded keys → lexicographic comparison is date comparison
   160→   160→      if (k >= cutoffKey) {
   161→   161→        const s = daily[k];
   162→   162→        recorded += 1;
   163→   163→        pomodoros += s.pomodoros;
   164→   164→        seconds += s.focusSeconds;
   165→   165→        if (s.pomodoros > 0) activeDays += 1;
   166→   166→      }
   167→   167→    });
   168→   168→    const hours = Math.floor(seconds / 3600);
   169→   169→    const minutes = Math.floor((seconds % 3600) / 60);
   170→   170→    return {
   171→   171→      pomodoros,
   172→   172→      activeDays,
   173→   173→      recorded,
   174→   174→      focusLabel:
   175→   175→        hours > 0 ? `${hours} h ${String(minutes).padStart(2, "0")} min` : `${minutes} min`,
   176→   176→    };
   177→   177→  }, [daily, period]);
   178→   178→
   179→   179→  const streaks = useMemo(() => computeStreaks(daily), [daily]);
   180→   180→
   181→   181→  /* ----- Current month grid ----- */
   182→   182→  const monthLabel = useMemo(() => {
   183→   183→    return capitalize(
   184→   184→      new Date(view.y, view.m, 1).toLocaleDateString("fr-FR", {
   185→   185→        month: "long",
   186→   186→        year: "numeric",
   187→   187→      }),
   188→   188→    );
   189→   189→  }, [view]);
   190→   190→
   191→   191→  const month = useMemo(() => {
   192→   192→    const now = new Date();
   193→   193→    const todayMid = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   194→   194→    const daysInMonth = new Date(view.y, view.m + 1, 0).getDate();
   195→   195→    // French weeks start Monday: JS Sunday=0 → Monday-first index
   196→   196→    const leading = (new Date(view.y, view.m, 1).getDay() + 6) % 7;
   197→   197→
   198→   198→    const cells: MonthCell[] = [];
   199→   199→    for (let d = 1; d <= daysInMonth; d++) {
   200→   200→      const key = dayKeyOf(view.y, view.m, d);
   201→   201→      const stat = daily[key] ?? EMPTY_STAT;
   202→   202→      const pomodoros = stat.pomodoros;
   203→   203→      const minutes = Math.floor(stat.focusSeconds / 60);
   204→   204→      const dateStr = new Date(view.y, view.m, d).toLocaleDateString("fr-FR", {
   205→   205→        day: "numeric",
   206→   206→        month: "long",
   207→   207→        year: "numeric",
   208→   208→      });
   209→   209→      cells.push({
   210→   210→        key,
   211→   211→        pomodoros,
   212→   212→        isFuture: new Date(view.y, view.m, d) > todayMid,
   213→   213→        isToday: key === todayKey(),
   214→   214→        title:
   215→   215→          pomodoros === 0
   216→   216→            ? `${dateStr} — aucun pomodoro`
   217→   217→            : `${dateStr} — ${pomodoros} pomodoro${pomodoros > 1 ? "s" : ""}, ${minutes} min de concentration`,
   218→   218→      });
   219→   219→    }
   220→   220→
   221→   221→    const monthPomodoros = cells.reduce((a, c) => a + c.pomodoros, 0);
   222→   222→    const trailing = (7 - ((leading + daysInMonth) % 7)) % 7;
   223→   223→    return { leading, trailing, cells, monthPomodoros };
   224→   224→  }, [daily, view]);
   225→   225→
   226→   226→  /* ----- Year heatmap (53 weeks × 7 days, Monday-first) ----- */
   227→   227→  const yearData = useMemo(() => {
   228→   228→    const now = new Date();
   229→   229→    const todayMid = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   230→   230→    const first = new Date(viewYear, 0, 1);
   231→   231→    // Start on the Monday on or before January 1st
   232→   232→    const start = new Date(first);
   233→   233→    start.setDate(first.getDate() - ((first.getDay() + 6) % 7));
   234→   234→    const end = new Date(viewYear, 11, 31);
   235→   235→
   236→   236→    const weeks: YearDayCell[][] = [];
   237→   237→    const cur = new Date(start);
   238→   238→    let yearPomodoros = 0;
   239→   239→    let yearActiveDays = 0;
   240→   240→
   241→   241→    while (cur <= end) {
   242→   242→      const week: YearDayCell[] = [];
   243→   243→      for (let i = 0; i < 7; i++) {
   244→   244→        const inYear = cur.getFullYear() === viewYear;
   245→   245→        const key = dayKeyOf(cur.getFullYear(), cur.getMonth(), cur.getDate());
   246→   246→        const pomodoros = (daily[key] ?? EMPTY_STAT).pomodoros;
   247→   247→        const isFuture = cur > todayMid;
   248→   248→        const isToday = key === todayKey();
   249→   249→        if (inYear && !isFuture && pomodoros > 0) {
   250→   250→          yearPomodoros += pomodoros;
   251→   251→          yearActiveDays += 1;
   252→   252→        }
   253→   253→        const minutes = Math.floor((daily[key] ?? EMPTY_STAT).focusSeconds / 60);
   254→   254→        const dateStr = new Date(cur).toLocaleDateString("fr-FR", {
   255→   255→          day: "numeric",
   256→   256→          month: "long",
   257→   257→          year: "numeric",
   258→   258→        });
   259→   259→        week.push({
   260→   260→          key,
   261→   261→          pomodoros,
   262→   262→          inYear,
   263→   263→          isFuture,
   264→   264→          isToday,
   265→   265→          title:
   266→   266→            !inYear || isFuture
   267→   267→              ? ""
   268→   268→              : pomodoros === 0
   269→   269→                ? `${dateStr} — aucun pomodoro`
   270→   270→                : `${dateStr} — ${pomodoros} pomodoro${pomodoros > 1 ? "s" : ""}, ${minutes} min de concentration`,
   271→   271→        });
   272→   272→        cur.setDate(cur.getDate() + 1);
   273→   273→      }
   274→   274→      weeks.push(week);
   275→   275→    }
   276→   276→
   277→   277→    // Month marks: week index of the 1st of each month, thinned to avoid overlap
   278→   278→    const marks: MonthMark[] = [];
   279→   279→    let lastWeek = -99;
   280→   280→    for (let m = 0; m < 12; m++) {
   281→   281→      const firstDay = new Date(viewYear, m, 1);
   282→   282→      const diff = Math.round((firstDay.getTime() - start.getTime()) / 86400000);
   283→   283→      const week = Math.floor(diff / 7);
   284→   284→      if (week - lastWeek >= 3) {
   285→   285→        marks.push({
   286→   286→          label: capitalize(
   287→   287→            firstDay.toLocaleDateString("fr-FR", { month: "short" }).replace(".", ""),
   288→   288→          ),
   289→   289→          week,
   290→   290→        });
   291→   291→        lastWeek = week;
   292→   292→      }
   293→   293→    }
   294→   294→
   295→   295→    return { weeks, yearPomodoros, yearActiveDays, marks };
   296→   296→  }, [daily, viewYear]);
   297→   297→
   298→   298→  /* ----- Weekday distribution (Monday-first) ----- */
   299→   299→  const weekday = useMemo(() => {
   300→   300→    const totals: number[] = Array(7).fill(0);
   301→   301→    Object.entries(daily).forEach(([k, v]) => {
   302→   302→      const parts = k.split("-");
   303→   303→      const d = new Date(Number(parts[0]), Number(parts[1]) - 1, Number(parts[2]));
   304→   304→      totals[(d.getDay() + 6) % 7] += v.pomodoros;
   305→   305→    });
   306→   306→    const max = Math.max(...totals);
   307→   307→    const bestIndex = totals.indexOf(max);
   308→   308→    return { totals, max, bestIndex };
   309→   309→  }, [daily]);
   310→   310→
   311→   311→  const now = new Date();
   312→   312→  const isCurrentMonth = view.y === now.getFullYear() && view.m === now.getMonth();
   313→   313→  const isCurrentYear = viewYear === now.getFullYear();
   314→   314→
   315→   315→  const goPrevMonth = () =>
   316→   316→    setView((v) => (v.m === 0 ? { y: v.y - 1, m: 11 } : { ...v, m: v.m - 1 }));
   317→   317→  const goNextMonth = () =>
   318→   318→    setView((v) => (v.m === 11 ? { y: v.y + 1, m: 0 } : { ...v, m: v.m + 1 }));
   319→   319→
   320→   320→  const gridAria = `${monthLabel} : ${month.monthPomodoros} ${
   321→   321→    month.monthPomodoros === 1 ? "pomodoro" : "pomodoros"
   322→   322→  } sur ${totals.pomodoros} ${totals.pomodoros === 1 ? "pomodoro" : "pomodoros"} enregistré${
   323→   323→    totals.pomodoros === 1 ? "" : "s"
   324→   324→  }`;
   325→   325→
   326→   326→  const yearAria = `Année ${viewYear} : ${yearData.yearPomodoros} ${
   327→   327→    yearData.yearPomodoros === 1 ? "pomodoro" : "pomodoros"
   328→   328→  } sur ${totals.pomodoros} ${totals.pomodoros === 1 ? "pomodoro" : "pomodoros"} enregistré${
   329→   329→    totals.pomodoros === 1 ? "" : "s"
   330→   330→  }`;
   331→   331→
   332→   332→  const hasData = totals.recorded > 0;
   333→   333→
   334→   334→  const handleExport = () => {
   335→   335→    if (Object.keys(daily).length === 0) {
   336→   336→      toast.info("Aucune donnée à exporter.");
   337→   337→      return;
   338→   338→    }
   339→   339→    downloadTextFile(dailyToCsv(daily), `focusly-statistiques-${todayKey()}.csv`);
   340→   340→    toast.success("Export CSV téléchargé.");
   341→   341→  };
   342→   342→
   343→   343→  /* ----- KPI cards (scoped to the selected period) ----- */
   344→   344→  const shown = periodTotals ?? totals;
   345→   345→  const periodLabel =
   346→   346→    period === null
   347→   347→      ? "Sur l’ensemble de votre historique."
   348→   348→      : `${period} derniers jours, aujourd’hui inclus.`;
   349→   349→
   350→   350→  const kpis: Array<{
   351→   351→    icon: LucideIcon;
   352→   352→    value: string;
   353→   353→    label: string;
   354→   354→    hint?: string;
   355→   355→    hint2?: string;
   356→   356→  }> = [
   357→   357→    {
   358→   358→      icon: Flame,
   359→   359→      value: String(shown.pomodoros),
   360→   360→      label: "Pomodoros",
   361→   361→    },
   362→   362→    {
   363→   363→      icon: Timer,
   364→   364→      value: shown.focusLabel,
   365→   365→      label: "Temps de concentration",
   366→   366→    },
   367→   367→    {
   368→   368→      icon: CalendarCheck,
   369→   369→      value: String(shown.activeDays),
   370→   370→      label: "Jours actifs",
   371→   371→      hint: `/ ${shown.recorded} jour${shown.recorded > 1 ? "s" : ""} enregistré${
   372→   372→        shown.recorded > 1 ? "s" : ""
   373→   373→      }`,
   374→   374→    },
   375→   375→    {
   376→   376→      // Streaks are date-independent: always computed on the whole history
   377→   377→      icon: Award,
   378→   378→      value: String(streaks.best),
   379→   379→      label: "Série record",
   380→   380→      hint: `actuelle : ${streaks.current} j`,
   381→   381→      hint2: period !== null ? "indépendante de la période" : undefined,
   382→   382→    },
   383→   383→  ];
   384→   384→
   385→   385→  return (
   386→   386→    <div className="mx-auto w-full max-w-[920px] px-5 py-8 sm:py-10">
   387→   387→      <Breadcrumb
   388→   388→        items={[{ label: "Accueil", href: "accueil" }, { label: "Statistiques" }]}
   389→   389→      />
   390→   390→
   391→   391→      <h1 className="text-3xl font-bold tracking-tight sm:text-4xl">Statistiques</h1>
   392→   392→      <p className="mt-3 max-w-[680px] text-[17px] text-muted-foreground">
   393→   393→        Vos calendriers de concentration, mois après mois et année après année. Vos
   394→   394→        données restent sur votre appareil.
   395→   395→      </p>
   396→   396→
   397→   397→      {!hasData && (
   398→   398→        <div className="mt-6 rounded-3xl border border-dashed p-8 text-center">
   399→   399→          <p className="text-[15px] text-soft">
   400→   400→            Aucune donnée pour l’instant. Terminez votre premier Pomodoro pour voir
   401→   401→            votre calendrier se remplir.
   402→   402→          </p>
   403→   403→        </div>
   404→   404→      )}
   405→   405→
   406→   406→      <div className="mt-8 flex flex-col gap-5">
   407→   407→        {/* ----- KPI row ----- */}
   408→   408→        <div className="flex flex-wrap items-center justify-between gap-3">
   409→   409→          <div>
   410→   410→            <h2 className="m-0 text-base font-bold tracking-tight">Vue d’ensemble</h2>
   411→   411→            <p className="mt-0.5 text-[13px] text-faint" aria-live="polite">
   412→   412→              {periodLabel}
   413→   413→            </p>
   414→   414→          </div>
   415→   415→          <div
   416→   416→            role="group"
   417→   417→            aria-label="Période des statistiques"
   418→   418→            className="flex rounded-xl border bg-secondary/60 p-1"
   419→   419→          >
   420→   420→            {PERIOD_OPTIONS.map((opt) => (
   421→   421→              <button
   422→   422→                key={opt.label}
   423→   423→                type="button"
   424→   424→                aria-pressed={period === opt.value}
   425→   425→                onClick={() => setPeriod(opt.value)}
   426→   426→                className={`min-h-8 whitespace-nowrap rounded-lg px-3 text-[12.5px] font-semibold transition-colors ${
   427→   427→                  period === opt.value
   428→   428→                    ? "bg-brand text-[#14161a] shadow-[0_2px_10px_-3px_var(--brand)]"
   429→   429→                    : "text-muted-foreground hover:text-foreground"
   430→   430→                }`}
   431→   431→              >
   432→   432→                {opt.label}
   433→   433→              </button>
   434→   434→            ))}
   435→   435→          </div>
   436→   436→        </div>
   437→   437→        <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
   438→   438→          {kpis.map((k) => (
   439→   439→            <div key={k.label} className="flex flex-col gap-2 rounded-3xl border bg-card p-5">
   440→   440→              <k.icon className="size-5 text-brand" aria-hidden />
   441→   441→              <span className="time-display text-2xl font-bold">{k.value}</span>
   442→   442→              <small className="text-[10px] font-semibold uppercase tracking-[0.09em] text-muted-foreground">
   443→   443→                {k.label}
   444→   444→              </small>
   445→   445→              {k.hint ? <span className="text-[11px] text-faint">{k.hint}</span> : null}
