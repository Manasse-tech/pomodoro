    69→function levelIndex(pomodoros: number): number {
    70→  if (pomodoros <= 0) return 0;
    71→  if (pomodoros <= 2) return 1;
    72→  if (pomodoros <= 4) return 2;
    73→  if (pomodoros <= 6) return 3;
    74→  return 4;
    75→}

    77→/** Local-date key — same decomposition as todayKey() in types.ts. */
    78→function dayKeyOf(y: number, m0: number, d: number): string {
    79→  return `${y}-${String(m0 + 1).padStart(2, "0")}-${String(d).padStart(2, "0")}`;
    80→}

    82→function capitalize(v: string): string {
    83→  return v.charAt(0).toUpperCase() + v.slice(1);

   119→export function StatistiquesView() {
   120→  const daily = useFocusly((s) => s.daily);
   121→  const [calendarTab, setCalendarTab] = useState<"mensuel" | "annuel">("mensuel");
   122→  const [view, setView] = useState<ViewMonth>(() => {
   123→    const n = new Date();
   124→    return { y: n.getFullYear(), m: n.getMonth() };
   125→  });
   126→  const [viewYear, setViewYear] = useState<number>(() => new Date().getFullYear());
   127→  /** Selected KPI period in days — null = all time (default, preserves old behavior). */
   128→  const [period, setPeriod] = useState<7 | 30 | 90 | null>(null);
   129→  /** Partage en image en cours (évite les doubles clics pendant la génération). */
   130→  const [sharePending, setSharePending] = useState(false);
   131→
   132→  /* ----- Grand totals across all recorded days ----- */
   133→  const totals = useMemo(() => {
   134→    let pomodoros = 0;
   135→    let seconds = 0;
   136→    let activeDays = 0;
   137→    const recorded = Object.keys(daily).length;
   138→    Object.values(daily).forEach((s) => {
   139→      pomodoros += s.pomodoros;
   140→      seconds += s.focusSeconds;
   141→      if (s.pomodoros > 0) activeDays += 1;
   142→    });
   143→    const hours = Math.floor(seconds / 3600);
   144→    const minutes = Math.floor((seconds % 3600) / 60);
   145→    return {
   146→      pomodoros,
   147→      activeDays,
   148→      recorded,
   149→      focusSeconds: seconds,
   150→      focusLabel:
   151→        hours > 0 ? `${hours} h ${String(minutes).padStart(2, "0")} min` : `${minutes} min`,
   152→    };
   153→  }, [daily]);
   154→
   155→  /* ----- Totals within the selected period (null when no filter) ----- */
   156→  const periodTotals = useMemo(() => {
   157→    if (period === null) return null;
   158→    const n = new Date();
   159→    // Cutoff = today minus (N-1) days → today is always included
   160→    const cutoff = new Date(n.getFullYear(), n.getMonth(), n.getDate());
   161→    cutoff.setDate(cutoff.getDate() - (period - 1));
   162→    const cutoffKey = todayKey(cutoff); // zero-padded YYYY-MM-DD (local)
   163→    let pomodoros = 0;
   164→    let seconds = 0;
   165→    let activeDays = 0;
   166→    let recorded = 0;
   167→    Object.keys(daily).forEach((k) => {
   168→      // Zero-padded keys → lexicographic comparison is date comparison
   169→      if (k >= cutoffKey) {
   170→        const s = daily[k];
   171→        recorded += 1;
   172→        pomodoros += s.pomodoros;
   173→        seconds += s.focusSeconds;
   174→        if (s.pomodoros > 0) activeDays += 1;
   175→      }
   176→    });
   177→    const hours = Math.floor(seconds / 3600);
   178→    const minutes = Math.floor((seconds % 3600) / 60);
   179→    return {
   180→      pomodoros,
   181→      activeDays,
   182→      recorded,
   183→      focusSeconds: seconds,
   184→      focusLabel:
   185→        hours > 0 ? `${hours} h ${String(minutes).padStart(2, "0")} min` : `${minutes} min`,
   186→    };
   187→  }, [daily, period]);
   188→
   189→  const streaks = useMemo(() => computeStreaks(daily), [daily]);
   190→
   191→  /* ----- Comparaison semaine vs semaine (lundi → dimanche, comme le reste de la page) ----- */
   192→  const weekCompare = useMemo(() => {
   193→    const n = new Date();
   194→    const todayMid = new Date(n.getFullYear(), n.getMonth(), n.getDate());
   195→    // Lundi de la semaine courante (dimanche = 0 → index lundi-first)
   196→    const monday = new Date(todayMid);
   197→    monday.setDate(monday.getDate() - ((monday.getDay() + 6) % 7));
   198→    const prevMonday = new Date(monday);
   199→    prevMonday.setDate(prevMonday.getDate() - 7);
   200→
   201→    // Somme des 7 clés de la semaine ; les jours futurs n’existent pas dans daily
   202→    const sumWeek = (start: Date) => {
   203→      let pomodoros = 0;
   204→      let seconds = 0;
   205→      const cur = new Date(start);
   206→      for (let i = 0; i < 7; i++) {
   207→        const s = daily[dayKeyOf(cur.getFullYear(), cur.getMonth(), cur.getDate())];
   208→        if (s) {
   209→          pomodoros += s.pomodoros;
   210→          seconds += s.focusSeconds;
   211→        }
   212→        cur.setDate(cur.getDate() + 1);
   213→      }
   214→      const hours = Math.floor(seconds / 3600);
   215→      const minutes = Math.floor((seconds % 3600) / 60);
   216→      return {
   217→        pomodoros,
   218→        focusLabel:
   219→          hours > 0 ? `${hours} h ${String(minutes).padStart(2, "0")} min` : `${minutes} min`,
   220→      };
   221→    };
   222→
   223→    const current = sumWeek(monday);
   224→    const previous = sumWeek(prevMonday);
   225→
   226→    // Position dans la semaine : 0 = lundi … 6 = dimanche
   227→    const dayIndex = (todayMid.getDay() + 6) % 7;
   228→    const daysElapsed = dayIndex + 1;
   229→
   230→    // Évolution en % — garde anti-division par zéro
   231→    let trend: "up" | "down" | "flat" | "new" | "empty" = "empty";
   232→    let pct = 0;
   233→    if (previous.pomodoros === 0 && current.pomodoros > 0) {
   234→      trend = "new";
   235→    } else if (previous.pomodoros > 0) {
   236→      pct = Math.round(((current.pomodoros - previous.pomodoros) / previous.pomodoros) * 100);
   237→      trend = pct > 0 ? "up" : pct < 0 ? "down" : "flat";
   238→    }
   239→
   240→    return { current, previous, daysElapsed, weekComplete: dayIndex === 6, trend, pct };
   241→  }, [daily]);
   242→
   243→  /* ----- Current month grid ----- */
   244→  const monthLabel = useMemo(() => {
   245→    return capitalize(
   246→      new Date(view.y, view.m, 1).toLocaleDateString("fr-FR", {
   247→        month: "long",
   248→        year: "numeric",
   249→      }),
   250→    );
   251→  }, [view]);
   252→
   253→  const month = useMemo(() => {
   254→    const now = new Date();
   255→    const todayMid = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   256→    const daysInMonth = new Date(view.y, view.m + 1, 0).getDate();
   257→    // French weeks start Monday: JS Sunday=0 → Monday-first index
   258→    const leading = (new Date(view.y, view.m, 1).getDay() + 6) % 7;
   259→
   260→    const cells: MonthCell[] = [];
   261→    for (let d = 1; d <= daysInMonth; d++) {
   262→      const key = dayKeyOf(view.y, view.m, d);
   263→      const stat = daily[key] ?? EMPTY_STAT;
   264→      const pomodoros = stat.pomodoros;
   265→      const minutes = Math.floor(stat.focusSeconds / 60);
   266→      const dateStr = new Date(view.y, view.m, d).toLocaleDateString("fr-FR", {
   267→        day: "numeric",
   268→        month: "long",
   269→        year: "numeric",
   270→      });
   271→      cells.push({
   272→        key,
   273→        pomodoros,
   274→        isFuture: new Date(view.y, view.m, d) > todayMid,
   275→        isToday: key === todayKey(),
   276→        title:
   277→          pomodoros === 0
   278→            ? `${dateStr} — aucun pomodoro`
   279→            : `${dateStr} — ${pomodoros} pomodoro${pomodoros > 1 ? "s" : ""}, ${minutes} min de concentration`,
   280→      });
   281→    }
   282→
   283→    const monthPomodoros = cells.reduce((a, c) => a + c.pomodoros, 0);
   284→    const trailing = (7 - ((leading + daysInMonth) % 7)) % 7;
   285→    return { leading, trailing, cells, monthPomodoros };
   286→  }, [daily, view]);
   287→
   288→  /* ----- Year heatmap (53 weeks × 7 days, Monday-first) ----- */
   289→  const yearData = useMemo(() => {
   290→    const now = new Date();
   291→    const todayMid = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   292→    const first = new Date(viewYear, 0, 1);
   293→    // Start on the Monday on or before January 1st
   294→    const start = new Date(first);
   295→    start.setDate(first.getDate() - ((first.getDay() + 6) % 7));
   296→    const end = new Date(viewYear, 11, 31);
   297→
   298→    const weeks: YearDayCell[][] = [];
   299→    const cur = new Date(start);
   300→    let yearPomodoros = 0;
   301→    let yearActiveDays = 0;
   302→
   303→    while (cur <= end) {
   304→      const week: YearDayCell[] = [];
   305→      for (let i = 0; i < 7; i++) {
   306→        const inYear = cur.getFullYear() === viewYear;
   307→        const key = dayKeyOf(cur.getFullYear(), cur.getMonth(), cur.getDate());
   308→        const pomodoros = (daily[key] ?? EMPTY_STAT).pomodoros;
   309→        const isFuture = cur > todayMid;
   310→        const isToday = key === todayKey();
   311→        if (inYear && !isFuture && pomodoros > 0) {
   312→          yearPomodoros += pomodoros;
   313→          yearActiveDays += 1;
   314→        }
   315→        const minutes = Math.floor((daily[key] ?? EMPTY_STAT).focusSeconds / 60);
   316→        const dateStr = new Date(cur).toLocaleDateString("fr-FR", {
   317→          day: "numeric",
   318→          month: "long",
   319→          year: "numeric",
   320→        });
   321→        week.push({
   322→          key,
   323→          pomodoros,
   324→          inYear,
   325→          isFuture,
   326→          isToday,
   327→          title:
   328→            !inYear || isFuture
   329→              ? ""
   330→              : pomodoros === 0
   331→                ? `${dateStr} — aucun pomodoro`
   332→                : `${dateStr} — ${pomodoros} pomodoro${pomodoros > 1 ? "s" : ""}, ${minutes} min de concentration`,
   333→        });
   334→        cur.setDate(cur.getDate() + 1);
   335→      }
   336→      weeks.push(week);
   337→    }
   338→
   339→    // Month marks: week index of the 1st of each month, thinned to avoid overlap
   340→    const marks: MonthMark[] = [];
   341→    let lastWeek = -99;
   342→    for (let m = 0; m < 12; m++) {
   343→      const firstDay = new Date(viewYear, m, 1);
   344→      const diff = Math.round((firstDay.getTime() - start.getTime()) / 86400000);
   345→      const week = Math.floor(diff / 7);
   346→      if (week - lastWeek >= 3) {
   347→        marks.push({
   348→          label: capitalize(
   349→            firstDay.toLocaleDateString("fr-FR", { month: "short" }).replace(".", ""),
   350→          ),
   351→          week,
   352→        });
   353→        lastWeek = week;
   354→      }
   355→    }
   356→
   357→    return { weeks, yearPomodoros, yearActiveDays, marks };
   358→  }, [daily, viewYear]);
   359→
   360→  /* ----- Weekday distribution (Monday-first) ----- */
   361→  const weekday = useMemo(() => {
   362→    const totals: number[] = Array(7).fill(0);
   363→    Object.entries(daily).forEach(([k, v]) => {
   364→      const parts = k.split("-");
   365→      const d = new Date(Number(parts[0]), Number(parts[1]) - 1, Number(parts[2]));
   366→      totals[(d.getDay() + 6) % 7] += v.pomodoros;
   367→    });
   368→    const max = Math.max(...totals);
   369→    const bestIndex = totals.indexOf(max);
   370→    return { totals, max, bestIndex };
   371→  }, [daily]);
   372→
   373→  const now = new Date();
   374→  const isCurrentMonth = view.y === now.getFullYear() && view.m === now.getMonth();
   375→  const isCurrentYear = viewYear === now.getFullYear();
   376→
   377→  const goPrevMonth = () =>
   378→    setView((v) => (v.m === 0 ? { y: v.y - 1, m: 11 } : { ...v, m: v.m - 1 }));
   379→  const goNextMonth = () =>
   380→    setView((v) => (v.m === 11 ? { y: v.y + 1, m: 0 } : { ...v, m: v.m + 1 }));
   381→
   382→  const gridAria = `${monthLabel} : ${month.monthPomodoros} ${
   383→    month.monthPomodoros === 1 ? "pomodoro" : "pomodoros"
   384→  } sur ${totals.pomodoros} ${totals.pomodoros === 1 ? "pomodoro" : "pomodoros"} enregistré${
   385→    totals.pomodoros === 1 ? "" : "s"
   386→  }`;
   387→
   388→  const yearAria = `Année ${viewYear} : ${yearData.yearPomodoros} ${
   389→    yearData.yearPomodoros === 1 ? "pomodoro" : "pomodoros"
   390→  } sur ${totals.pomodoros} ${totals.pomodoros === 1 ? "pomodoro" : "pomodoros"} enregistré${
   391→    totals.pomodoros === 1 ? "" : "s"
   392→  }`;
   393→
   394→  const hasData = totals.recorded > 0;
   395→
   396→  /* ----- Comparaison hebdomadaire : libellés dérivés ----- */
   397→  const weekMax = Math.max(weekCompare.current.pomodoros, weekCompare.previous.pomodoros);
   398→  const weekAria = `Cette semaine : ${weekCompare.current.pomodoros} ${
   399→    weekCompare.current.pomodoros === 1 ? "pomodoro" : "pomodoros"
   400→  }. Semaine dernière : ${weekCompare.previous.pomodoros} ${
   401→    weekCompare.previous.pomodoros === 1 ? "pomodoro" : "pomodoros"
   402→  }.`;
   403→  // Largeur de barre (max des deux semaines = pleine largeur, plancher 2 % si actif)
   404→  const weekBarPct = (v: number) =>
   405→    weekMax > 0 ? Math.max((v / weekMax) * 100, 2) : 0;
   406→
   407→  const handleExport = () => {
   408→    if (Object.keys(daily).length === 0) {
   409→      toast.info("Aucune donnée à exporter.");
   410→      return;
   411→    }
   412→    downloadTextFile(dailyToCsv(daily), `focusly-statistiques-${todayKey()}.csv`);
   413→    toast.success("Export CSV téléchargé.");
   414→  };
   415→
   416→  /* ----- KPI cards (scoped to the selected period) ----- */
   417→  const shown = periodTotals ?? totals;
   418→  const periodLabel =
   419→    period === null
   420→      ? "Sur l’ensemble de votre historique."
   421→      : `${period} derniers jours, aujourd’hui inclus.`;
   422→  // Libellé condensé pour l’image partagée (pas de phrase complète)
   423→  const imagePeriodLabel = period === null ? "Historique complet" : `${period} derniers jours`;
   424→
   425→  const kpis: Array<{
   426→    icon: LucideIcon;
   427→    value: string;
   428→    label: string;
   429→    hint?: string;
   430→    hint2?: string;
   431→  }> = [
   432→    {
   433→      icon: Flame,
   434→      value: String(shown.pomodoros),
   435→      label: "Pomodoros",
   436→    },
   437→    {
   438→      icon: Timer,
   439→      value: shown.focusLabel,
   440→      label: "Temps de concentration",
   441→    },
   442→    {
   443→      icon: CalendarCheck,
   444→      value: String(shown.activeDays),
   445→      label: "Jours actifs",
   446→      hint: `/ ${shown.recorded} jour${shown.recorded > 1 ? "s" : ""} enregistré${
   447→        shown.recorded > 1 ? "s" : ""
   448→      }`,
   449→    },
   450→    {
   451→      // Streaks are date-independent: always computed on the whole history
   452→      icon: Award,
   453→      value: String(streaks.best),
   454→      label: "Série record",
   455→      hint: `actuelle : ${streaks.current} j`,
   456→      hint2: period !== null ? "indépendante de la période" : undefined,
   457→    },
   458→  ];
   459→
   460→  const handleShareImage = async () => {
   461→    setSharePending(true);
   462→    try {
   463→      const result = await shareOrDownloadStatsImage({
   464→        pomodoros: shown.pomodoros,
   465→        focusSeconds: shown.focusSeconds,
   466→        bestStreak: streaks.best,
   467→        periodLabel: imagePeriodLabel,
   468→        fileName: `focusly-statistiques-${todayKey()}.png`,
   469→      });
   470→      // shared : l’OS affiche déjà son retour ; cancelled : annulation silencieuse
   471→      if (result === "downloaded") toast.success("Image générée.");
   472→    } catch {
   473→      toast.error("Impossible de générer l’image.");
   474→    } finally {
   475→      setSharePending(false);
   476→    }
   477→  };
   478→
   479→  return (
   480→    <div className="mx-auto w-full max-w-[920px] px-5 py-8 sm:py-10">
   481→      <Breadcrumb
   482→        items={[{ label: "Accueil", href: "accueil" }, { label: "Statistiques" }]}
   483→      />
   484→
   485→      <h1 className="text-3xl font-bold tracking-tight sm:text-4xl">Statistiques</h1>
   486→      <p className="mt-3 max-w-[680px] text-[17px] text-muted-foreground">
   487→        Vos calendriers de concentration, mois après mois et année après année. Vos
   488→        données restent sur votre appareil.
   489→      </p>
   490→
   491→      {!hasData && (
   492→        <div className="mt-6 rounded-3xl border border-dashed p-8 text-center">
   493→          <p className="text-[15px] text-soft">
   494→            Aucune donnée pour l’instant. Terminez votre premier Pomodoro pour voir
   495→            votre calendrier se remplir.
   496→          </p>
   497→        </div>
   498→      )}
   499→
   500→      <div className="mt-8 flex flex-col gap-5">
   501→        {/* ----- KPI row ----- */}
   502→        <div className="flex flex-wrap items-center justify-between gap-3">
   503→          <div>
   504→            <h2 className="m-0 text-base font-bold tracking-tight">Vue d’ensemble</h2>
   505→            <p className="mt-0.5 text-[13px] text-faint" aria-live="polite">
   506→              {periodLabel}
   507→            </p>
   508→          </div>
   509→          <div
   510→            role="group"
   511→            aria-label="Période des statistiques"
   512→            className="flex rounded-xl border bg-secondary/60 p-1"
   513→          >
   514→            {PERIOD_OPTIONS.map((opt) => (
   515→              <button
   516→                key={opt.label}
   517→                type="button"
   518→                aria-pressed={period === opt.value}
   519→                onClick={() => setPeriod(opt.value)}
   520→                className={`min-h-8 whitespace-nowrap rounded-lg px-3 text-[12.5px] font-semibold transition-colors ${
   521→                  period === opt.value
   522→                    ? "bg-brand text-[#14161a] shadow-[0_2px_10px_-3px_var(--brand)]"
   523→                    : "text-muted-foreground hover:text-foreground"
   524→                }`}
   525→              >
