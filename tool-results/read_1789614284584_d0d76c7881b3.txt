    89→    89→/** Sum of pomodoros over every day of a month — the same derivation as the
    90→    90→ * monthly calendar (local dayKeyOf keys, EMPTY_STAT fallback); shared by the
    91→    91→ * calendar and the monthly goal gauge so the math lives in one place. */
    92→    92→function monthPomodorosOf(daily: Record<string, DailyStat>, y: number, m0: number): number {
    93→    93→  const days = daysInMonthOf(y, m0);
    94→    94→  let sum = 0;
    95→    95→  for (let d = 1; d <= days; d++) {
    96→    96→    sum += (daily[dayKeyOf(y, m0, d)] ?? EMPTY_STAT).pomodoros;
    97→    97→  }
    98→    98→  return sum;
    99→    99→}

   434→   434→

   436→   436→  const isCurrentMonth = view.y === now.getFullYear() && view.m === now.getMonth();
   437→   437→  const isCurrentYear = viewYear === now.getFullYear();
   438→   438→

   440→   440→    setView((v) => (v.m === 0 ? { y: v.y - 1, m: 11 } : { ...v, m: v.m - 1 }));
