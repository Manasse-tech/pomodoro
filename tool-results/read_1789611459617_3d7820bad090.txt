   121→   121→export function StatistiquesView() {
   122→   122→  const daily = useFocusly((s) => s.daily);
   123→   123→  const settings = useFocusly((s) => s.settings);
   124→   124→  const [calendarTab, setCalendarTab] = useState<"mensuel" | "annuel">("mensuel");
   125→   125→  const [view, setView] = useState<ViewMonth>(() => {
   126→   126→    const n = new Date();
   127→   127→    return { y: n.getFullYear(), m: n.getMonth() };
   128→   128→  });
   129→   129→  const [viewYear, setViewYear] = useState<number>(() => new Date().getFullYear());
   130→   130→  /** Selected KPI period in days — null = all time (default, preserves old behavior). */
   131→   131→  const [period, setPeriod] = useState<7 | 30 | 90 | null>(null);
   132→   132→  /** Partage en image en cours (évite les doubles clics pendant la génération). */
   133→   133→  const [sharePending, setSharePending] = useState(false);
   134→   134→
   135→   135→  /* ----- Grand totals across all recorded days ----- */
   136→   136→  const totals = useMemo(() => {
   137→   137→    let pomodoros = 0;
   138→   138→    let seconds = 0;
   139→   139→    let activeDays = 0;
   140→   140→    const recorded = Object.keys(daily).length;
   141→   141→    Object.values(daily).forEach((s) => {
   142→   142→      pomodoros += s.pomodoros;
   143→   143→      seconds += s.focusSeconds;
   144→   144→      if (s.pomodoros > 0) activeDays += 1;
   145→   145→    });
   146→   146→    const hours = Math.floor(seconds / 3600);
   147→   147→    const minutes = Math.floor((seconds % 3600) / 60);
   148→   148→    return {
   149→   149→      pomodoros,
   150→   150→      activeDays,
   151→   151→      recorded,
   152→   152→      focusSeconds: seconds,
   153→   153→      focusLabel:
   154→   154→        hours > 0 ? `${hours} h ${String(minutes).padStart(2, "0")} min` : `${minutes} min`,
   155→   155→    };
   156→   156→  }, [daily]);
   157→   157→
   158→   158→  /* ----- Totals within the selected period (null when no filter) ----- */
   159→   159→  const periodTotals = useMemo(() => {
   160→   160→    if (period === null) return null;
   161→   161→    const n = new Date();
   162→   162→    // Cutoff = today minus (N-1) days → today is always included
   163→   163→    const cutoff = new Date(n.getFullYear(), n.getMonth(), n.getDate());
   164→   164→    cutoff.setDate(cutoff.getDate() - (period - 1));
   165→   165→    const cutoffKey = todayKey(cutoff); // zero-padded YYYY-MM-DD (local)
   166→   166→    let pomodoros = 0;
   167→   167→    let seconds = 0;
   168→   168→    let activeDays = 0;
   169→   169→    let recorded = 0;
   170→   170→    Object.keys(daily).forEach((k) => {
   171→   171→      // Zero-padded keys → lexicographic comparison is date comparison
   172→   172→      if (k >= cutoffKey) {
   173→   173→        const s = daily[k];
   174→   174→        recorded += 1;
   175→   175→        pomodoros += s.pomodoros;
   176→   176→        seconds += s.focusSeconds;
   177→   177→        if (s.pomodoros > 0) activeDays += 1;
   178→   178→      }
   179→   179→    });
   180→   180→    const hours = Math.floor(seconds / 3600);
   181→   181→    const minutes = Math.floor((seconds % 3600) / 60);
   182→   182→    return {
   183→   183→      pomodoros,
   184→   184→      activeDays,
   185→   185→      recorded,
   186→   186→      focusSeconds: seconds,
   187→   187→      focusLabel:
   188→   188→        hours > 0 ? `${hours} h ${String(minutes).padStart(2, "0")} min` : `${minutes} min`,

   199→   199→    const monday = new Date(todayMid);
   200→   200→    monday.setDate(monday.getDate() - ((monday.getDay() + 6) % 7));
   201→   201→    const prevMonday = new Date(monday);
   202→   202→    prevMonday.setDate(prevMonday.getDate() - 7);
   203→   203→
   204→   204→    // Somme des 7 clés de la semaine ; les jours futurs n’existent pas dans daily

   206→   206→      let pomodoros = 0;
   207→   207→      let seconds = 0;
   208→   208→      const cur = new Date(start);
   209→   209→      for (let i = 0; i < 7; i++) {
   210→   210→        const s = daily[dayKeyOf(cur.getFullYear(), cur.getMonth(), cur.getDate())];
   211→   211→        if (s) {
   212→   212→          pomodoros += s.pomodoros;
   213→   213→          seconds += s.focusSeconds;
   214→   214→        }
   215→   215→        cur.setDate(cur.getDate() + 1);
   216→   216→      }
   217→   217→      const hours = Math.floor(seconds / 3600);
   218→   218→      const minutes = Math.floor((seconds % 3600) / 60);
   219→   219→      return {
   220→   220→        pomodoros,
   221→   221→        focusLabel:
   222→   222→          hours > 0 ? `${hours} h ${String(minutes).padStart(2, "0")} min` : `${minutes} min`,
   223→   223→      };
   224→   224→    };
   225→   225→
   226→   226→    const current = sumWeek(monday);
   227→   227→    const previous = sumWeek(prevMonday);
   228→   228→
   229→   229→    // Position dans la semaine : 0 = lundi … 6 = dimanche
   230→   230→    const dayIndex = (todayMid.getDay() + 6) % 7;
   231→   231→    const daysElapsed = dayIndex + 1;
   232→   232→
   233→   233→    // Évolution en % — garde anti-division par zéro
   234→   234→    let trend: "up" | "down" | "flat" | "new" | "empty" = "empty";
   235→   235→    let pct = 0;
   236→   236→    if (previous.pomodoros === 0 && current.pomodoros > 0) {
   237→   237→      trend = "new";
   238→   238→    } else if (previous.pomodoros > 0) {
   239→   239→      pct = Math.round(((current.pomodoros - previous.pomodoros) / previous.pomodoros) * 100);
   240→   240→      trend = pct > 0 ? "up" : pct < 0 ? "down" : "flat";
   241→   241→    }
   242→   242→
   243→   243→    return { current, previous, daysElapsed, weekComplete: dayIndex === 6, trend, pct };

   249→   249→    const goal = perDay * 7;
   250→   250→    const done = weekCompare.current.pomodoros;
   251→   251→    // Lundi = jour 1 … dimanche = 7 (même convention lundi-first que weekCompare)
   252→   252→    const daysElapsed = weekCompare.daysElapsed;
   253→   253→    return {
   254→   254→      perDay,
   255→   255→      goal,
   256→   256→      done,
   257→   257→      progress: goal > 0 ? Math.min(1, done / goal) : 0,
   258→   258→      remaining: Math.max(0, goal - done),
   259→   259→      daysLeft: 7 - daysElapsed,
   260→   260→      onPace: done >= perDay * daysElapsed,
   261→   261→    };
   262→   262→  }, [settings.dailyGoal, weekCompare]);
   263→   263→
   264→   264→  /* ----- Current month grid ----- */
   265→   265→  const monthLabel = useMemo(() => {
   266→   266→    return capitalize(
   267→   267→      new Date(view.y, view.m, 1).toLocaleDateString("fr-FR", {
   268→   268→        month: "long",
   269→   269→        year: "numeric",
   270→   270→      }),
   271→   271→    );
   272→   272→  }, [view]);
   273→   273→
   274→   274→  const month = useMemo(() => {
   275→   275→    const now = new Date();
   276→   276→    const todayMid = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   277→   277→    const daysInMonth = new Date(view.y, view.m + 1, 0).getDate();
   278→   278→    // French weeks start Monday: JS Sunday=0 → Monday-first index
   279→   279→    const leading = (new Date(view.y, view.m, 1).getDay() + 6) % 7;
   280→   280→
   281→   281→    const cells: MonthCell[] = [];
   282→   282→    for (let d = 1; d <= daysInMonth; d++) {
   283→   283→      const key = dayKeyOf(view.y, view.m, d);
   284→   284→      const stat = daily[key] ?? EMPTY_STAT;
   285→   285→      const pomodoros = stat.pomodoros;
   286→   286→      const minutes = Math.floor(stat.focusSeconds / 60);
   287→   287→      const dateStr = new Date(view.y, view.m, d).toLocaleDateString("fr-FR", {
   288→   288→        day: "numeric",
   289→   289→        month: "long",
   290→   290→        year: "numeric",
   291→   291→      });
   292→   292→      cells.push({
   293→   293→        key,
   294→   294→        pomodoros,

   297→   297→        title:
   298→   298→          pomodoros === 0
   299→   299→            ? `${dateStr} — aucun pomodoro`
   300→   300→            : `${dateStr} — ${pomodoros} pomodoro${pomodoros > 1 ? "s" : ""}, ${minutes} min de concentration`,
   301→   301→      });
   302→   302→    }
   303→   303→
   304→   304→    const monthPomodoros = cells.reduce((a, c) => a + c.pomodoros, 0);
   305→   305→    const trailing = (7 - ((leading + daysInMonth) % 7)) % 7;
   306→   306→    return { leading, trailing, cells, monthPomodoros };
   307→   307→  }, [daily, view]);
   308→   308→
   309→   309→  /* ----- Year heatmap (53 weeks × 7 days, Monday-first) ----- */
   310→   310→  const yearData = useMemo(() => {
   311→   311→    const now = new Date();
   312→   312→    const todayMid = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   313→   313→    const first = new Date(viewYear, 0, 1);
   314→   314→    // Start on the Monday on or before January 1st

   316→   316→    start.setDate(first.getDate() - ((first.getDay() + 6) % 7));

   319→   319→    const weeks: YearDayCell[][] = [];

   337→   337→        const dateStr = new Date(cur).toLocaleDateString("fr-FR", {
   338→   338→          day: "numeric",
   339→   339→          month: "long",
   340→   340→          year: "numeric",
   341→   341→        });

   343→   343→          key,
   344→   344→          pomodoros,
   345→   345→          inYear,
   346→   346→          isFuture,
   347→   347→          isToday,

   349→   349→            !inYear || isFuture

   351→   351→              : pomodoros === 0

   354→   354→        });
   355→   355→        cur.setDate(cur.getDate() + 1);
   356→   356→      }
   357→   357→      weeks.push(week);
   358→   358→    }
   359→   359→
   360→   360→    // Month marks: week index of the 1st of each month, thinned to avoid overlap

   366→   366→      const week = Math.floor(diff / 7);
   367→   367→      if (week - lastWeek >= 3) {
   368→   368→        marks.push({
   369→   369→          label: capitalize(
   370→   370→            firstDay.toLocaleDateString("fr-FR", { month: "short" }).replace(".", ""),
   371→   371→          ),
   372→   372→          week,
   373→   373→        });
   374→   374→        lastWeek = week;
   375→   375→      }
   376→   376→    }
   377→   377→
   378→   378→    return { weeks, yearPomodoros, yearActiveDays, marks };
   379→   379→  }, [daily, viewYear]);
   380→   380→
   381→   381→  /* ----- Weekday distribution (Monday-first) ----- */
   382→   382→  const weekday = useMemo(() => {
   383→   383→    const totals: number[] = Array(7).fill(0);
   384→   384→    Object.entries(daily).forEach(([k, v]) => {
   385→   385→      const parts = k.split("-");
   386→   386→      const d = new Date(Number(parts[0]), Number(parts[1]) - 1, Number(parts[2]));
   387→   387→      totals[(d.getDay() + 6) % 7] += v.pomodoros;
   388→   388→    });
   389→   389→    const max = Math.max(...totals);
   390→   390→    const bestIndex = totals.indexOf(max);
   391→   391→    return { totals, max, bestIndex };
   392→   392→  }, [daily]);
   393→   393→
   394→   394→  const now = new Date();
   395→   395→  const isCurrentMonth = view.y === now.getFullYear() && view.m === now.getMonth();
   396→   396→  const isCurrentYear = viewYear === now.getFullYear();
   397→   397→
   398→   398→  const goPrevMonth = () =>
   399→   399→    setView((v) => (v.m === 0 ? { y: v.y - 1, m: 11 } : { ...v, m: v.m - 1 }));
   400→   400→  const goNextMonth = () =>
   401→   401→    setView((v) => (v.m === 11 ? { y: v.y + 1, m: 0 } : { ...v, m: v.m + 1 }));
   402→   402→
   403→   403→  const gridAria = `${monthLabel} : ${month.monthPomodoros} ${
   404→   404→    month.monthPomodoros === 1 ? "pomodoro" : "pomodoros"
   405→   405→  } sur ${totals.pomodoros} ${totals.pomodoros === 1 ? "pomodoro" : "pomodoros"} enregistré${
   406→   406→    totals.pomodoros === 1 ? "" : "s"
   407→   407→  }`;
   408→   408→
   409→   409→  const yearAria = `Année ${viewYear} : ${yearData.yearPomodoros} ${
   410→   410→    yearData.yearPomodoros === 1 ? "pomodoro" : "pomodoros"
   411→   411→  } sur ${totals.pomodoros} ${totals.pomodoros === 1 ? "pomodoro" : "pomodoros"} enregistré${
   412→   412→    totals.pomodoros === 1 ? "" : "s"
   413→   413→  }`;
   414→   414→
   415→   415→  const hasData = totals.recorded > 0;
   416→   416→
   417→   417→  /* ----- Comparaison hebdomadaire : libellés dérivés ----- */
   418→   418→  const weekMax = Math.max(weekCompare.current.pomodoros, weekCompare.previous.pomodoros);
   419→   419→  const weekAria = `Cette semaine : ${weekCompare.current.pomodoros} ${
   420→   420→    weekCompare.current.pomodoros === 1 ? "pomodoro" : "pomodoros"
   421→   421→  }. Semaine dernière : ${weekCompare.previous.pomodoros} ${
   422→   422→    weekCompare.previous.pomodoros === 1 ? "pomodoro" : "pomodoros"
   423→   423→  }.`;
   424→   424→  // Largeur de barre (max des deux semaines = pleine largeur, plancher 2 % si actif)
   425→   425→  const weekBarPct = (v: number) =>
   426→   426→    weekMax > 0 ? Math.max((v / weekMax) * 100, 2) : 0;
   427→   427→  // Largeur de la jauge d’objectif (plancher 2 % si active, même convention)
   428→   428→  const goalPct = weekGoal.done > 0 ? Math.max(weekGoal.progress * 100, 2) : 0;
   429→   429→
   430→   430→  const handleExport = () => {
   431→   431→    if (Object.keys(daily).length === 0) {
   432→   432→      toast.info("Aucune donnée à exporter.");
   433→   433→      return;
