    41→    41→/** Full French date from a local YYYY-MM-DD key — built from local parts, never UTC */
    42→    42→function dueDateLabel(key: string): string {
    43→    43→  const [y, m, d] = key.split("-").map(Number);
    44→    44→  if (!y || !m || !d) return key;
    45→    45→  return frDate(new Date(y, m - 1, d).getTime());
    46→    46→}

    57→    57→/** Compact recurrence selector — « Aucune / Quotid. / Ouvrés / Hebdo / X j » chips (add form + inline edit) */
    58→    58→function RecurrencePicker({
    59→    59→  value,
    60→    60→  onChange,
    61→    61→  label,
    62→    62→  days,
    63→    63→  onDaysChange,
    64→    64→}: {
    65→    65→  value: TaskRecurrence | null;
    66→    66→  onChange: (next: TaskRecurrence | null) => void;
    67→    67→  label: string;
    68→    68→  /** Interval in days for the « custom » recurrence (2–365) */
    69→    69→  days: number;
    70→    70→  onDaysChange: (next: number) => void;
    71→    71→}) {
    72→    72→  /* Raw text while typing — partial input (« 1 » en route to « 12 ») must not be clamped mid-keystroke */
    73→    73→  const [rawDays, setRawDays] = useState<string | null>(null);
    74→    74→
    75→    75→  /* Derived from RECURRENCE_SHORT so every recurrence kind gets its pill automatically */
    76→    76→  const options: Array<{ value: TaskRecurrence | null; label: string }> = [
    77→    77→    { value: null, label: "Aucune" },
    78→    78→    ...(Object.entries(RECURRENCE_SHORT) as Array<[TaskRecurrence, string]>).map(
    79→    79→      ([v, l]) => ({
    80→    80→        value: v,
    81→    81→        // Once selected, the custom pill echoes the live interval (e.g. « 3 j »)
    82→    82→        label: v === "custom" && value === "custom" ? `${days} j` : l,
    83→    83→      }),
    84→    84→    ),
    85→    85→  ];
    86→    86→
    87→    87→  const commitDays = (text: string) => {
    88→    88→    const n = Number.parseInt(text, 10);
    89→    89→    if (Number.isFinite(n)) onDaysChange(clampRecurrenceDays(n));
    90→    90→  };
    91→    91→
    92→    92→  return (
    93→    93→    <div role="group" aria-label={label} className="flex flex-wrap items-center gap-1.5">
    94→    94→      <span aria-hidden className="text-[11px] text-faint">
    95→    95→        Répétition
    96→    96→      </span>
    97→    97→      {options.map((o) => (
    98→    98→        <button
    99→    99→          key={o.value ?? "none"}
   100→   100→          type="button"
   101→   101→          aria-pressed={value === o.value}
   102→   102→          onClick={() => onChange(o.value)}
   103→   103→          className={`rounded-full border px-2.5 py-1 text-xs transition-colors ${
   104→   104→            value === o.value
   105→   105→              ? "border-brand/40 bg-brand/15 font-medium text-brand"
   106→   106→              : "text-muted-foreground hover:text-foreground"
   107→   107→          }`}
   108→   108→        >
   109→   109→          {o.label}
   110→   110→        </button>
   111→   111→      ))}
   112→   112→      {value === "custom" && (
   113→   113→        <label
   114→   114→          title="Répéter tous les N jours (2 à 365)"
   115→   115→          className="inline-flex items-center gap-1 rounded-full border border-brand/40 bg-brand/15 px-2.5 py-1 text-xs font-medium text-brand"
   116→   116→        >
   117→   117→          <span aria-hidden>Tous les</span>
   118→   118→          <input
   119→   119→            type="number"
   120→   120→            inputMode="numeric"
   121→   121→            min={RECURRENCE_DAYS_MIN}
   122→   122→            max={RECURRENCE_DAYS_MAX}
   123→   123→            step={1}
   124→   124→            value={rawDays ?? String(days)}
   125→   125→            aria-label="Nombre de jours entre chaque répétition, entre 2 et 365"
   126→   126→            onChange={(e) => {
   127→   127→              setRawDays(e.target.value);
   128→   128→              commitDays(e.target.value);
   129→   129→            }}
   130→   130→            onBlur={() => setRawDays(null)}
   131→   131→            className="tnum w-11 rounded-full bg-transparent text-center text-xs font-semibold text-brand outline-none [appearance:textfield] focus-visible:ring-2 focus-visible:ring-brand/60 [&::-webkit-inner-spin-button]:appearance-none [&::-webkit-outer-spin-button]:appearance-none"
   132→   132→          />
   133→   133→          <span aria-hidden>jours</span>
   134→   134→        </label>
   135→   135→      )}
   136→   136→    </div>
   137→   137→  );
   138→   138→}
