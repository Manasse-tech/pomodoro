    55→/* Types & constants                                                   */

    57→
    58→interface ContactMsg {
    59→  id: string;
    60→  name: string;
    61→  email: string;
    62→  subject: string | null;
    63→  message: string;
    64→  reply: string | null;
    65→  repliedAt: string | null;
    66→  read: boolean;
    67→  archived: boolean;
    68→  createdAt: string;
    69→}
    70→
    71→/** sessionStorage key — holds the admin key for the current tab only. */
    72→const STORAGE_KEY = "focusly.adminKey";
    73→
    74→type AdminFilter = "actifs" | "non-lus" | "archives";
    75→
    76→const FILTERS: Array<{ value: AdminFilter; label: string }> = [
    77→  { value: "actifs", label: "Actifs" },
    78→  { value: "non-lus", label: "Non lus" },
    79→  { value: "archives", label: "Archivés" },
    80→];
    81→
    82→/** Nombre de messages par page (pagination serveur via ?page=&pageSize=). */
    83→const PAGE_SIZE = 25;
    84→
    85→/** Longueur maximale d'une réponse, alignée sur la limite d'un message. */
    86→const REPLY_MAX_LENGTH = 5000;
    87→
    88→/** Normalise un message renvoyé par l'API (champs optionnels → valeurs sûres). */
    89→function normalizeMsg(m: ContactMsg): ContactMsg {
    90→  return {
    91→    ...m,
    92→    read: m.read ?? false,
    93→    archived: m.archived ?? false,
    94→    reply: m.reply ?? null,
    95→    repliedAt: m.repliedAt ?? null,
    96→  };
    97→}
    98→
    99→/** Brouillon de réponse pré-rempli (français) — le « … » est à personnaliser. */
   100→function buildReplyTemplate(msg: ContactMsg): string {
   101→  return [
   102→    `Bonjour ${msg.name},`,
   103→    "",
   104→    "Merci pour votre message. …",
   105→    "",
   106→    "Cordialement,",
   107→    "L’équipe Focusly",
   108→  ].join("\n");
   109→}
   110→
   111→/** Lien mailto pré-rempli avec la réponse éditée — le client de messagerie prend le relais. */
   112→function buildMailtoHref(msg: ContactMsg, body: string): string {
   113→  const subject = `RE: ${msg.subject ?? "Votre message Focusly"}`;
   114→  return `mailto:${msg.email}?subject=${encodeURIComponent(subject)}&body=${encodeURIComponent(body)}`;
   115→}

   120→
   121→export function AdminView() {
   122→  const [adminKey, setAdminKey] = useState<string | null>(null);
   123→  const [entered, setEntered] = useState("");
   124→  const [gateError, setGateError] = useState<string | null>(null);
   125→  const [unlocking, setUnlocking] = useState(false);
   126→  const [messages, setMessages] = useState<ContactMsg[]>([]);
   127→  const [loading, setLoading] = useState(false);
   128→  const [filter, setFilter] = useState<AdminFilter>("actifs");
   129→  const [query, setQuery] = useState("");
   130→  /* Recherche serveur (?q=) — valeur saisie débouncée (~300 ms) */
   131→  const [search, setSearch] = useState("");
   132→  /* Pagination serveur */
   133→  const [page, setPage] = useState(1);
   134→  const [total, setTotal] = useState(0);
   135→  const [pageSize, setPageSize] = useState(PAGE_SIZE);
   136→  const [unreadCount, setUnreadCount] = useState(0);
   137→  /* Réponse aux messages */
   138→  const [replyTarget, setReplyTarget] = useState<ContactMsg | null>(null);
   139→  const [replyDraft, setReplyDraft] = useState("");
   140→  const [sendingReply, setSendingReply] = useState(false);
   141→  const replyTextareaRef = useRef<HTMLTextAreaElement | null>(null);
   142→  const [exportingCsv, setExportingCsv] = useState(false);
   143→
   144→  /* Restore a previously unlocked key (same tab only) */
   145→  useEffect(() => {
   146→    const stored = window.sessionStorage.getItem(STORAGE_KEY);
   147→    if (stored) setAdminKey(stored);
   148→  }, []);
   149→
   150→  /** Garde anti-course : seule la réponse de la dernière requête émise s'applique. */
   151→  const loadSeq = useRef(0);
   152→  const loadMessages = useCallback(
   153→    async (key: string, targetPage: number, q: string, f: AdminFilter) => {
   154→      const seq = ++loadSeq.current;
   155→      setLoading(true);
   156→      try {
   157→        const params = new URLSearchParams({
   158→          page: String(targetPage),
   159→          pageSize: String(PAGE_SIZE),
   160→          filter: f,
   161→        });
   162→        if (q !== "") params.set("q", q);
   163→        const res = await fetch(`/api/contact?${params.toString()}`, {
   164→          headers: { "x-admin-key": key },
   165→          cache: "no-store",
   166→        });
   167→        if (seq !== loadSeq.current) return; // requête plus récente en vol
   168→        if (!res.ok) {
   169→          const data = (await res.json().catch(() => null)) as { error?: string } | null;
   170→          if (res.status === 401) {
   171→            // The stored key is no longer valid → back to the gate
   172→            window.sessionStorage.removeItem(STORAGE_KEY);
   173→            setAdminKey(null);
   174→            setGateError("Clé incorrecte.");
   175→            return;
   176→          }
   177→          toast.error(data?.error ?? "Erreur réseau.");
   178→          return;
   179→        }
   180→        const data = (await res.json()) as {
   181→          items?: ContactMsg[];
   182→          total?: number;
   183→          page?: number;
   184→          pageSize?: number;
   185→          unread?: number;
   186→        };
   187→        if (seq !== loadSeq.current) return; // requête plus récente en vol
   188→        const items = (data.items ?? []).map(normalizeMsg);
   189→        setMessages(items);
   190→        setTotal(typeof data.total === "number" ? data.total : items.length);
   191→        setPageSize(typeof data.pageSize === "number" ? data.pageSize : PAGE_SIZE);
   192→        setUnreadCount(typeof data.unread === "number" ? data.unread : 0);
   193→        // Le serveur ramène la page demandée dans les bornes valides.
   194→        if (typeof data.page === "number" && data.page !== targetPage) {
   195→          setPage(data.page);
   196→        }
   197→      } catch {
   198→        if (seq !== loadSeq.current) return;
   199→        toast.error("Erreur réseau.");
   200→      } finally {
   201→        if (seq === loadSeq.current) setLoading(false);
   202→      }
   203→    },
   204→    [],
   205→  );
   206→
   207→  /* Debounce the raw search input (~300 ms) before it reaches the server as ?q= */
   208→  useEffect(() => {
   209→    const timer = window.setTimeout(() => setSearch(query.trim()), 300);
   210→    return () => window.clearTimeout(timer);
   211→  }, [query]);
   212→
   213→  /* Load (or reload) on key/page/search/filter change. A search or filter change
   214→     resets the pagination to page 1 within the same commit (single fetch) — the
   215→     signature guard skips the redundant refetch caused by the page reset, while
   216→     the refresh button (direct loadMessages call) stays unaffected. */
   217→  const prevQueryRef = useRef({ search: "", filter: "actifs" as AdminFilter });
   218→  const lastFetchRef = useRef<string | null>(null);
   219→  useEffect(() => {
   220→    if (!adminKey) {
   221→      // Déverrouillage futur = première charge garantie (401 ou déconnexion).
   222→      lastFetchRef.current = null;
   223→      return;
   224→    }
   225→    let targetPage = page;
   226→    if (prevQueryRef.current.search !== search || prevQueryRef.current.filter !== filter) {
   227→      prevQueryRef.current = { search, filter };
   228→      if (page !== 1) {
   229→        targetPage = 1;
   230→        setPage(1);
   231→      }
   232→    }
   233→    const signature = `${adminKey}|${targetPage}|${search}|${filter}`;
   234→    if (lastFetchRef.current === signature) return;
   235→    lastFetchRef.current = signature;
   236→    void loadMessages(adminKey, targetPage, search, filter);
   237→  }, [adminKey, page, search, filter, loadMessages]);
   238→
   239→  /* ----- Gate: validate the key then remember it for the tab ----- */
   240→  const handleUnlock = async (e: React.FormEvent<HTMLFormElement>) => {
   241→    e.preventDefault();
   242→    const k = entered.trim();
   243→    if (k.length === 0) {
   244→      setGateError("Veuillez saisir la clé d’administration.");
   245→      return;
   246→    }
   247→    setUnlocking(true);
   248→    setGateError(null);
   249→    try {
   250→      const res = await fetch(`/api/contact?page=1&pageSize=${PAGE_SIZE}`, {
   251→        headers: { "x-admin-key": k },
   252→        cache: "no-store",
   253→      });
   254→      if (res.ok) {
   255→        window.sessionStorage.setItem(STORAGE_KEY, k);
   256→        setAdminKey(k);
   257→        setPage(1);
   258→        setEntered(""); // never keep the key in the DOM
   259→      } else if (res.status === 401) {
   260→        setGateError("Clé incorrecte.");
   261→      } else if (res.status === 429) {
   262→        setGateError("Trop de tentatives, réessayez plus tard.");
   263→      } else {
   264→        const data = (await res.json().catch(() => null)) as { error?: string } | null;
   265→        setGateError(data?.error ?? "Erreur réseau.");
   266→      }
   267→    } catch {
   268→      setGateError("Erreur réseau.");
   269→    } finally {
   270→      setUnlocking(false);
   271→    }
   272→  };
   273→
   274→  /* ----- Sign out: clear the key from storage and state ----- */
   275→  const handleSignOut = () => {
   276→    window.sessionStorage.removeItem(STORAGE_KEY);
   277→    setAdminKey(null);
   278→    setMessages([]);
   279→    setEntered("");
   280→    setGateError(null);
   281→    setFilter("actifs");
   282→    setQuery("");
   283→    setSearch("");
   284→    prevQueryRef.current = { search: "", filter: "actifs" };
   285→    setPage(1);
   286→    setTotal(0);
   287→    setUnreadCount(0);
   288→    setReplyTarget(null);
   289→    setReplyDraft("");
   290→  };
   291→
   292→  /* ----- Server-side filters: a message that no longer matches the active chip
   293→     leaves the list optimistically. Last item of a page > 1 → previous page (refetch). ----- */
   294→  const dropFromList = (id: string) => {
   295→    if (messages.length <= 1 && page > 1) {
   296→      setPage((p) => Math.max(1, p - 1));
   297→      return;
   298→    }
   299→    setMessages((ms) => ms.filter((m) => m.id !== id));
   300→    setTotal((t) => Math.max(0, t - 1));
   301→  };
   302→
   303→  /* ----- Toggle read state (optimistic, reverted on failure) ----- */
   304→  const toggleRead = async (msg: ContactMsg) => {
   305→    const next = !msg.read;
   306→    setMessages((ms) => ms.map((m) => (m.id === msg.id ? { ...m, read: next } : m)));
   307→    // Le compteur « non lus » ne compte que les messages actifs (non archivés).
   308→    if (!msg.archived) setUnreadCount((c) => Math.max(0, c + (next ? -1 : 1)));
   309→    try {
   310→      const res = await fetch("/api/contact", {
   311→        method: "PATCH",
   312→        headers: {
   313→          "Content-Type": "application/json",
   314→          "x-admin-key": adminKey ?? "",
   315→        },
   316→        body: JSON.stringify({ id: msg.id, read: next }),
   317→      });
   318→      if (!res.ok) {
   319→        const data = (await res.json().catch(() => null)) as { error?: string } | null;
   320→        throw new Error(data?.error ?? "Erreur réseau.");
   321→      }
   322→      toast.success("Message mis à jour.");
   323→      // Filtre serveur « non-lus » : le message lu ne correspond plus → il sort de la liste.
   324→      if (next && filter === "non-lus") dropFromList(msg.id);
   325→    } catch (err) {
   326→      setMessages((ms) =>
   327→        ms.map((m) => (m.id === msg.id ? { ...m, read: msg.read } : m)),
   328→      );
   329→      if (!msg.archived) setUnreadCount((c) => Math.max(0, c + (msg.read ? -1 : 1)));
   330→      toast.error(err instanceof Error ? err.message : "Erreur réseau.");
   331→    }
   332→  };
   333→
   334→  /* ----- Archive / restore (optimistic, reverted on failure) ----- */
   335→  const toggleArchived = async (msg: ContactMsg) => {
   336→    const next = !msg.archived;
   337→    setMessages((ms) =>
   338→      ms.map((m) => (m.id === msg.id ? { ...m, archived: next } : m)),
   339→    );
   340→    // Archiver un message non lu le retire du compteur « non lus » (et inversement).
   341→    if (!msg.read) setUnreadCount((c) => Math.max(0, c + (next ? -1 : 1)));
   342→    try {
   343→      const res = await fetch("/api/contact", {
   344→        method: "PATCH",
   345→        headers: {
   346→          "Content-Type": "application/json",
   347→          "x-admin-key": adminKey ?? "",
   348→        },
   349→        body: JSON.stringify({ id: msg.id, archived: next }),
   350→      });
   351→      if (!res.ok) {
   352→        const data = (await res.json().catch(() => null)) as { error?: string } | null;
   353→        throw new Error(data?.error ?? "Erreur réseau.");
   354→      }
   355→      toast.success(next ? "Message archivé." : "Message restauré.");
   356→      // Filtres serveur : un message qui ne correspond plus au filtre actif sort de la liste.
   357→      if (filter === "actifs" && next) dropFromList(msg.id);
   358→      else if (filter === "archives" && !next) dropFromList(msg.id);
   359→    } catch (err) {
   360→      setMessages((ms) =>
   361→        ms.map((m) => (m.id === msg.id ? { ...m, archived: msg.archived } : m)),
   362→      );
   363→      if (!msg.read) setUnreadCount((c) => Math.max(0, c + (msg.archived ? -1 : 1)));
   364→      toast.error(err instanceof Error ? err.message : "Erreur réseau.");
   365→    }
   366→  };
   367→
   368→  /* ----- Delete a message (optimistic, re-inserted on failure) ----- */
   369→  const handleDelete = async (msg: ContactMsg) => {
   370→    const index = messages.findIndex((m) => m.id === msg.id);
   371→    setMessages((ms) => ms.filter((m) => m.id !== msg.id));
   372→    setTotal((t) => Math.max(0, t - 1));
   373→    if (!msg.read && !msg.archived) setUnreadCount((c) => Math.max(0, c - 1));
   374→    try {
   375→      const res = await fetch(`/api/contact?id=${encodeURIComponent(msg.id)}`, {
   376→        method: "DELETE",
   377→        headers: { "x-admin-key": adminKey ?? "" },
   378→      });
   379→      if (!res.ok) {
   380→        const data = (await res.json().catch(() => null)) as { error?: string } | null;
   381→        throw new Error(data?.error ?? "Erreur réseau.");
   382→      }
   383→      toast.success("Message supprimé.");
   384→      // Dernier message de la page (hors page 1) → revenir à la page précédente.
   385→      if (messages.length <= 1 && page > 1) setPage((p) => Math.max(1, p - 1));
   386→    } catch (err) {
   387→      setMessages((ms) => {
   388→        const copy = [...ms];
   389→        copy.splice(Math.max(0, index < 0 ? copy.length : index), 0, msg);
   390→        return copy;
   391→      });
   392→      setTotal((t) => t + 1);
   393→      if (!msg.read && !msg.archived) setUnreadCount((c) => c + 1);
   394→      toast.error(err instanceof Error ? err.message : "Erreur réseau.");
   395→    }
   396→  };
   397→
   398→  /* ----- Reply dialog: save the answer, then hand over to the mail client ----- */
   399→  const openReplyDialog = (msg: ContactMsg) => {
   400→    setReplyTarget(msg);
   401→    // Déjà répondu → on repart de la réponse existante, sinon brouillon type.
   402→    setReplyDraft(msg.reply ?? buildReplyTemplate(msg));
   403→  };
   404→
   405→  const handleReplyOpenChange = (open: boolean) => {
   406→    if (!open && !sendingReply) {
   407→      setReplyTarget(null);
   408→      setReplyDraft("");
   409→    }
   410→  };
   411→
   412→  const handleSendReply = async () => {
   413→    const target = replyTarget;
   414→    if (!target) return;
   415→    const trimmed = replyDraft.trim();
   416→    if (trimmed.length === 0 || trimmed.length > REPLY_MAX_LENGTH) return;
   417→    setSendingReply(true);
   418→    try {
   419→      const res = await fetch("/api/contact", {
   420→        method: "PATCH",
   421→        headers: {
   422→          "Content-Type": "application/json",
   423→          "x-admin-key": adminKey ?? "",
   424→        },
   425→        body: JSON.stringify({ id: target.id, reply: trimmed }),
   426→      });
   427→      if (!res.ok) {
   428→        const data = (await res.json().catch(() => null)) as { error?: string } | null;
   429→        throw new Error(data?.error ?? "Erreur réseau.");
   430→      }
   431→      // (1) + (2) la réponse est enregistrée et le message marqué comme répondu.
   432→      setMessages((ms) =>
   433→        ms.map((m) =>
   434→          m.id === target.id
   435→            ? { ...m, reply: trimmed, repliedAt: new Date().toISOString(), read: true }
   436→            : m,
   437→        ),
   438→      );
   439→      if (!target.read) setUnreadCount((c) => Math.max(0, c - 1));
   440→      // Filtre serveur « non-lus » : le message répondu (donc lu) sort de la liste.
   441→      if (!target.read && filter === "non-lus") dropFromList(target.id);
   442→      // (3) le client de messagerie prend le relais avec le brouillon édité.
   443→      window.location.href = buildMailtoHref(target, trimmed);
   444→      // (4) confirmation
   445→      toast.success("Réponse enregistrée.");
   446→      setReplyDraft("");
   447→      setReplyTarget((current) => (current?.id === target.id ? null : current));
   448→    } catch (err) {
   449→      toast.error(err instanceof Error ? err.message : "Erreur réseau.");
   450→    } finally {
   451→      setSendingReply(false);
   452→    }
   453→  };
   454→
   455→  /* ----- Export ALL messages matching the current search + filter as CSV
   456→     (server-side endpoint ?format=csv — no pagination limit) ----- */
   457→  const handleExportCsv = async () => {
   458→    if (exportingCsv) return;
   459→    if (total === 0) {
   460→      toast.info("Aucun message à exporter.");
   461→      return;
   462→    }
   463→    setExportingCsv(true);
   464→    try {
   465→      const params = new URLSearchParams({ format: "csv", filter });
   466→      if (search !== "") params.set("q", search);
   467→      const res = await fetch(`/api/contact?${params.toString()}`, {
   468→        headers: { "x-admin-key": adminKey ?? "" },
   469→        cache: "no-store",
   470→      });
   471→      if (res.status === 401) {
   472→        // The stored key is no longer valid → back to the gate
   473→        window.sessionStorage.removeItem(STORAGE_KEY);
   474→        setAdminKey(null);
   475→        setGateError("Clé incorrecte.");
   476→        return;
   477→      }
   478→      if (!res.ok) {
   479→        const data = (await res.json().catch(() => null)) as { error?: string } | null;
   480→        throw new Error(data?.error ?? "Erreur réseau.");
   481→      }
   482→      const blob = await res.blob();
   483→      // Nom de fichier proposé par le serveur (Content-Disposition), sinon repli local.
   484→      const disposition = res.headers.get("Content-Disposition") ?? "";
   485→      const filename =
   486→        /filename="([^"]+)"/.exec(disposition)?.[1] ?? `focusly-messages-${todayKey()}.csv`;
   487→      const url = URL.createObjectURL(blob);
   488→      const anchor = document.createElement("a");
   489→      anchor.href = url;
   490→      anchor.download = filename;
   491→      document.body.appendChild(anchor);
   492→      anchor.click();
   493→      document.body.removeChild(anchor);
   494→      URL.revokeObjectURL(url);
   495→      toast.success("Export CSV téléchargé.");
   496→    } catch (err) {
   497→      toast.error(err instanceof Error ? err.message : "Erreur réseau.");
   498→    } finally {
   499→      setExportingCsv(false);
   500→    }
   501→  };
   502→
   503→  /* ---------------------------------------------------------------- */
   504→  /* Gate (no valid key in this tab)                                   */
   505→  /* ---------------------------------------------------------------- */
   506→
   507→  if (!adminKey) {
   508→    return (
   509→      <section className="mx-auto w-full max-w-[920px] px-5 py-8 sm:py-10">
   510→        <Breadcrumb
   511→          items={[{ label: "Accueil", href: "accueil" }, { label: "Administration" }]}
   512→        />
   513→        <div className="mx-auto mt-6 w-full max-w-md rounded-3xl border bg-card p-8">
   514→          <div
   515→            className="mx-auto flex size-12 items-center justify-center rounded-2xl bg-brand/15"
   516→            aria-hidden
   517→          >
   518→            <Lock className="size-6 text-brand" />
   519→          </div>
   520→          <h1 className="mt-4 text-center text-2xl font-bold tracking-tight">
   521→            Administration
   522→          </h1>
   523→          <p className="mt-2 text-center text-[15px] text-muted-foreground">
   524→            Espace réservé à l’équipe Focusly.
   525→          </p>
   526→
   527→          <form onSubmit={handleUnlock} className="mt-6 flex flex-col gap-3">
   528→            <label htmlFor="admin-key" className="text-sm font-medium text-soft">
   529→              Clé d’administration
