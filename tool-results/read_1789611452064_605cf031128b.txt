    71→function levelIndex(pomodoros: number): number {
    72→  if (pomodoros <= 0) return 0;
    73→  if (pomodoros <= 2) return 1;
    74→  if (pomodoros <= 4) return 2;
    75→  if (pomodoros <= 6) return 3;
    76→  return 4;
    77→}

   121→export function StatistiquesView() {
   122→  const daily = useFocusly((s) => s.daily);
   123→  const settings = useFocusly((s) => s.settings);
   124→  const [calendarTab, setCalendarTab] = useState<"mensuel" | "annuel">("mensuel");
   125→  const [view, setView] = useState<ViewMonth>(() => {
   126→    const n = new Date();
   127→    return { y: n.getFullYear(), m: n.getMonth() };
   128→  });
   129→  const [viewYear, setViewYear] = useState<number>(() => new Date().getFullYear());
   130→  /** Selected KPI period in days — null = all time (default, preserves old behavior). */
   131→  const [period, setPeriod] = useState<7 | 30 | 90 | null>(null);
   132→  /** Partage en image en cours (évite les doubles clics pendant la génération). */
   133→  const [sharePending, setSharePending] = useState(false);
   134→
   135→  /* ----- Grand totals across all recorded days ----- */
   136→  const totals = useMemo(() => {
   137→    let pomodoros = 0;
   138→    let seconds = 0;
   139→    let activeDays = 0;
   140→    const recorded = Object.keys(daily).length;
   141→    Object.values(daily).forEach((s) => {
   142→      pomodoros += s.pomodoros;
   143→      seconds += s.focusSeconds;
   144→      if (s.pomodoros > 0) activeDays += 1;
   145→    });
   146→    const hours = Math.floor(seconds / 3600);
   147→    const minutes = Math.floor((seconds % 3600) / 60);
   148→    return {
   149→      pomodoros,
   150→      activeDays,
   151→      recorded,
   152→      focusSeconds: seconds,
   153→      focusLabel:
   154→        hours > 0 ? `${hours} h ${String(minutes).padStart(2, "0")} min` : `${minutes} min`,
   155→    };
   156→  }, [daily]);
   157→
   158→  /* ----- Totals within the selected period (null when no filter) ----- */
   159→  const periodTotals = useMemo(() => {
   160→    if (period === null) return null;
   161→    const n = new Date();
   162→    // Cutoff = today minus (N-1) days → today is always included
   163→    const cutoff = new Date(n.getFullYear(), n.getMonth(), n.getDate());
   164→    cutoff.setDate(cutoff.getDate() - (period - 1));
   165→    const cutoffKey = todayKey(cutoff); // zero-padded YYYY-MM-DD (local)
   166→    let pomodoros = 0;
   167→    let seconds = 0;
   168→    let activeDays = 0;
   169→    let recorded = 0;
   170→    Object.keys(daily).forEach((k) => {
   171→      // Zero-padded keys → lexicographic comparison is date comparison
   172→      if (k >= cutoffKey) {
   173→        const s = daily[k];
   174→        recorded += 1;
   175→        pomodoros += s.pomodoros;
   176→        seconds += s.focusSeconds;
   177→        if (s.pomodoros > 0) activeDays += 1;
   178→      }
   179→    });
   180→    const hours = Math.floor(seconds / 3600);
   181→    const minutes = Math.floor((seconds % 3600) / 60);
   182→    return {
   183→      pomodoros,
   184→      activeDays,
   185→      recorded,
   186→      focusSeconds: seconds,
   187→      focusLabel:
   188→        hours > 0 ? `${hours} h ${String(minutes).padStart(2, "0")} min` : `${minutes} min`,
   189→    };
   190→  }, [daily, period]);
   191→
   192→  const streaks = useMemo(() => computeStreaks(daily), [daily]);
   193→
   194→  /* ----- Comparaison semaine vs semaine (lundi → dimanche, comme le reste de la page) ----- */
   195→  const weekCompare = useMemo(() => {
   196→    const n = new Date();
   197→    const todayMid = new Date(n.getFullYear(), n.getMonth(), n.getDate());
   198→    // Lundi de la semaine courante (dimanche = 0 → index lundi-first)
   199→    const monday = new Date(todayMid);
   200→    monday.setDate(monday.getDate() - ((monday.getDay() + 6) % 7));
   201→    const prevMonday = new Date(monday);
   202→    prevMonday.setDate(prevMonday.getDate() - 7);
   203→
   204→    // Somme des 7 clés de la semaine ; les jours futurs n’existent pas dans daily
   205→    const sumWeek = (start: Date) => {
   206→      let pomodoros = 0;
   207→      let seconds = 0;
   208→      const cur = new Date(start);
   209→      for (let i = 0; i < 7; i++) {
   210→        const s = daily[dayKeyOf(cur.getFullYear(), cur.getMonth(), cur.getDate())];
   211→        if (s) {
   212→          pomodoros += s.pomodoros;
   213→          seconds += s.focusSeconds;
   214→        }
   215→        cur.setDate(cur.getDate() + 1);
   216→      }
   217→      const hours = Math.floor(seconds / 3600);
   218→      const minutes = Math.floor((seconds % 3600) / 60);
   219→      return {
   220→        pomodoros,
   221→        focusLabel:
   222→          hours > 0 ? `${hours} h ${String(minutes).padStart(2, "0")} min` : `${minutes} min`,
   223→      };
   224→    };
   225→
   226→    const current = sumWeek(monday);
   227→    const previous = sumWeek(prevMonday);
   228→
   229→    // Position dans la semaine : 0 = lundi … 6 = dimanche
   230→    const dayIndex = (todayMid.getDay() + 6) % 7;
   231→    const daysElapsed = dayIndex + 1;
   232→
   233→    // Évolution en % — garde anti-division par zéro
   234→    let trend: "up" | "down" | "flat" | "new" | "empty" = "empty";
   235→    let pct = 0;
   236→    if (previous.pomodoros === 0 && current.pomodoros > 0) {
   237→      trend = "new";
   238→    } else if (previous.pomodoros > 0) {
   239→      pct = Math.round(((current.pomodoros - previous.pomodoros) / previous.pomodoros) * 100);
   240→      trend = pct > 0 ? "up" : pct < 0 ? "down" : "flat";
   241→    }
   242→
   243→    return { current, previous, daysElapsed, weekComplete: dayIndex === 6, trend, pct };
   244→  }, [daily]);
   245→
   246→  /* ----- Jauge d’objectif hebdomadaire (dérivée : réglages × semaine courante) ----- */
   247→  const weekGoal = useMemo(() => {
   248→    const perDay = settings.dailyGoal;
   249→    const goal = perDay * 7;
   250→    const done = weekCompare.current.pomodoros;
   251→    // Lundi = jour 1 … dimanche = 7 (même convention lundi-first que weekCompare)
   252→    const daysElapsed = weekCompare.daysElapsed;
   253→    return {
   254→      perDay,
   255→      goal,
   256→      done,
   257→      progress: goal > 0 ? Math.min(1, done / goal) : 0,
   258→      remaining: Math.max(0, goal - done),
   259→      daysLeft: 7 - daysElapsed,
   260→      onPace: done >= perDay * daysElapsed,
   261→    };
   262→  }, [settings.dailyGoal, weekCompare]);
   263→
   264→  /* ----- Current month grid ----- */
   265→  const monthLabel = useMemo(() => {
   266→    return capitalize(
   267→      new Date(view.y, view.m, 1).toLocaleDateString("fr-FR", {
   268→        month: "long",
   269→        year: "numeric",
   270→      }),
   271→    );
   272→  }, [view]);
   273→
   274→  const month = useMemo(() => {
   275→    const now = new Date();
   276→    const todayMid = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   277→    const daysInMonth = new Date(view.y, view.m + 1, 0).getDate();
   278→    // French weeks start Monday: JS Sunday=0 → Monday-first index
   279→    const leading = (new Date(view.y, view.m, 1).getDay() + 6) % 7;
   280→
   281→    const cells: MonthCell[] = [];
   282→    for (let d = 1; d <= daysInMonth; d++) {
   283→      const key = dayKeyOf(view.y, view.m, d);
   284→      const stat = daily[key] ?? EMPTY_STAT;
   285→      const pomodoros = stat.pomodoros;
   286→      const minutes = Math.floor(stat.focusSeconds / 60);
   287→      const dateStr = new Date(view.y, view.m, d).toLocaleDateString("fr-FR", {
   288→        day: "numeric",
   289→        month: "long",
   290→        year: "numeric",
   291→      });
   292→      cells.push({
   293→        key,
   294→        pomodoros,
   295→        isFuture: new Date(view.y, view.m, d) > todayMid,
   296→        isToday: key === todayKey(),
   297→        title:
   298→          pomodoros === 0
   299→            ? `${dateStr} — aucun pomodoro`
   300→            : `${dateStr} — ${pomodoros} pomodoro${pomodoros > 1 ? "s" : ""}, ${minutes} min de concentration`,
   301→      });
   302→    }
   303→
   304→    const monthPomodoros = cells.reduce((a, c) => a + c.pomodoros, 0);
   305→    const trailing = (7 - ((leading + daysInMonth) % 7)) % 7;
   306→    return { leading, trailing, cells, monthPomodoros };
   307→  }, [daily, view]);
   308→
   309→  /* ----- Year heatmap (53 weeks × 7 days, Monday-first) ----- */
   310→  const yearData = useMemo(() => {
   311→    const now = new Date();
   312→    const todayMid = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   313→    const first = new Date(viewYear, 0, 1);
   314→    // Start on the Monday on or before January 1st
   315→    const start = new Date(first);
   316→    start.setDate(first.getDate() - ((first.getDay() + 6) % 7));
   317→    const end = new Date(viewYear, 11, 31);
   318→
   319→    const weeks: YearDayCell[][] = [];
   320→    const cur = new Date(start);
   321→    let yearPomodoros = 0;
   322→    let yearActiveDays = 0;
   323→
   324→    while (cur <= end) {
   325→      const week: YearDayCell[] = [];
   326→      for (let i = 0; i < 7; i++) {
   327→        const inYear = cur.getFullYear() === viewYear;
   328→        const key = dayKeyOf(cur.getFullYear(), cur.getMonth(), cur.getDate());
   329→        const pomodoros = (daily[key] ?? EMPTY_STAT).pomodoros;
   330→        const isFuture = cur > todayMid;
   331→        const isToday = key === todayKey();
   332→        if (inYear && !isFuture && pomodoros > 0) {
   333→          yearPomodoros += pomodoros;
   334→          yearActiveDays += 1;
   335→        }
   336→        const minutes = Math.floor((daily[key] ?? EMPTY_STAT).focusSeconds / 60);
   337→        const dateStr = new Date(cur).toLocaleDateString("fr-FR", {
   338→          day: "numeric",
   339→          month: "long",
   340→          year: "numeric",
   341→        });
   342→        week.push({
   343→          key,
   344→          pomodoros,
   345→          inYear,
   346→          isFuture,
   347→          isToday,
   348→          title:
   349→            !inYear || isFuture
   350→              ? ""
   351→              : pomodoros === 0
   352→                ? `${dateStr} — aucun pomodoro`
   353→                : `${dateStr} — ${pomodoros} pomodoro${pomodoros > 1 ? "s" : ""}, ${minutes} min de concentration`,
   354→        });
   355→        cur.setDate(cur.getDate() + 1);
   356→      }
   357→      weeks.push(week);
   358→    }
   359→
   360→    // Month marks: week index of the 1st of each month, thinned to avoid overlap
   361→    const marks: MonthMark[] = [];
   362→    let lastWeek = -99;
   363→    for (let m = 0; m < 12; m++) {
   364→      const firstDay = new Date(viewYear, m, 1);
   365→      const diff = Math.round((firstDay.getTime() - start.getTime()) / 86400000);
   366→      const week = Math.floor(diff / 7);
   367→      if (week - lastWeek >= 3) {
   368→        marks.push({
   369→          label: capitalize(
   370→            firstDay.toLocaleDateString("fr-FR", { month: "short" }).replace(".", ""),
   371→          ),
   372→          week,
   373→        });
   374→        lastWeek = week;
   375→      }
   376→    }
   377→
   378→    return { weeks, yearPomodoros, yearActiveDays, marks };
   379→  }, [daily, viewYear]);
   380→
   381→  /* ----- Weekday distribution (Monday-first) ----- */
   382→  const weekday = useMemo(() => {
   383→    const totals: number[] = Array(7).fill(0);
   384→    Object.entries(daily).forEach(([k, v]) => {
   385→      const parts = k.split("-");
   386→      const d = new Date(Number(parts[0]), Number(parts[1]) - 1, Number(parts[2]));
   387→      totals[(d.getDay() + 6) % 7] += v.pomodoros;
   388→    });
   389→    const max = Math.max(...totals);
   390→    const bestIndex = totals.indexOf(max);
   391→    return { totals, max, bestIndex };
   392→  }, [daily]);
   393→
   394→  const now = new Date();
   395→  const isCurrentMonth = view.y === now.getFullYear() && view.m === now.getMonth();
   396→  const isCurrentYear = viewYear === now.getFullYear();
   397→
   398→  const goPrevMonth = () =>
   399→    setView((v) => (v.m === 0 ? { y: v.y - 1, m: 11 } : { ...v, m: v.m - 1 }));
   400→  const goNextMonth = () =>
   401→    setView((v) => (v.m === 11 ? { y: v.y + 1, m: 0 } : { ...v, m: v.m + 1 }));
   402→
   403→  const gridAria = `${monthLabel} : ${month.monthPomodoros} ${
   404→    month.monthPomodoros === 1 ? "pomodoro" : "pomodoros"
   405→  } sur ${totals.pomodoros} ${totals.pomodoros === 1 ? "pomodoro" : "pomodoros"} enregistré${
   406→    totals.pomodoros === 1 ? "" : "s"
   407→  }`;
   408→
   409→  const yearAria = `Année ${viewYear} : ${yearData.yearPomodoros} ${
   410→    yearData.yearPomodoros === 1 ? "pomodoro" : "pomodoros"
   411→  } sur ${totals.pomodoros} ${totals.pomodoros === 1 ? "pomodoro" : "pomodoros"} enregistré${
   412→    totals.pomodoros === 1 ? "" : "s"
   413→  }`;
   414→
   415→  const hasData = totals.recorded > 0;
   416→
   417→  /* ----- Comparaison hebdomadaire : libellés dérivés ----- */
   418→  const weekMax = Math.max(weekCompare.current.pomodoros, weekCompare.previous.pomodoros);
   419→  const weekAria = `Cette semaine : ${weekCompare.current.pomodoros} ${
   420→    weekCompare.current.pomodoros === 1 ? "pomodoro" : "pomodoros"
   421→  }. Semaine dernière : ${weekCompare.previous.pomodoros} ${
   422→    weekCompare.previous.pomodoros === 1 ? "pomodoro" : "pomodoros"
   423→  }.`;
   424→  // Largeur de barre (max des deux semaines = pleine largeur, plancher 2 % si actif)
   425→  const weekBarPct = (v: number) =>
   426→    weekMax > 0 ? Math.max((v / weekMax) * 100, 2) : 0;
   427→  // Largeur de la jauge d’objectif (plancher 2 % si active, même convention)
   428→  const goalPct = weekGoal.done > 0 ? Math.max(weekGoal.progress * 100, 2) : 0;
   429→
   430→  const handleExport = () => {
   431→    if (Object.keys(daily).length === 0) {
   432→      toast.info("Aucune donnée à exporter.");
   433→      return;
   434→    }
   435→    downloadTextFile(dailyToCsv(daily), `focusly-statistiques-${todayKey()}.csv`);
   436→    toast.success("Export CSV téléchargé.");
   437→  };
   438→
   439→  /* ----- KPI cards (scoped to the selected period) ----- */
   440→  const shown = periodTotals ?? totals;
   441→  const periodLabel =
   442→    period === null
   443→      ? "Sur l’ensemble de votre historique."
   444→      : `${period} derniers jours, aujourd’hui inclus.`;
   445→  // Libellé condensé pour l’image partagée (pas de phrase complète)
   446→  const imagePeriodLabel = period === null ? "Historique complet" : `${period} derniers jours`;
   447→
   448→  const kpis: Array<{
   449→    icon: LucideIcon;
   450→    value: string;
   451→    label: string;
   452→    hint?: string;
   453→    hint2?: string;
   454→  }> = [
   455→    {
   456→      icon: Flame,
   457→      value: String(shown.pomodoros),
   458→      label: "Pomodoros",
   459→    },
   460→    {
   461→      icon: Timer,
   462→      value: shown.focusLabel,
   463→      label: "Temps de concentration",
   464→    },
   465→    {
   466→      icon: CalendarCheck,
   467→      value: String(shown.activeDays),
   468→      label: "Jours actifs",
   469→      hint: `/ ${shown.recorded} jour${shown.recorded > 1 ? "s" : ""} enregistré${
   470→        shown.recorded > 1 ? "s" : ""
   471→      }`,
   472→    },
   473→    {
   474→      // Streaks are date-independent: always computed on the whole history
   475→      icon: Award,
   476→      value: String(streaks.best),
   477→      label: "Série record",
   478→      hint: `actuelle : ${streaks.current} j`,
   479→      hint2: period !== null ? "indépendante de la période" : undefined,
   480→    },
   481→  ];
   482→
   483→  const handleShareImage = async () => {
   484→    setSharePending(true);
   485→    try {
   486→      const result = await shareOrDownloadStatsImage({
   487→        pomodoros: shown.pomodoros,
   488→        focusSeconds: shown.focusSeconds,
   489→        bestStreak: streaks.best,
   490→        periodLabel: imagePeriodLabel,
   491→        fileName: `focusly-statistiques-${todayKey()}.png`,
   492→      });
   493→      // shared : l’OS affiche déjà son retour ; cancelled : annulation silencieuse
   494→      if (result === "downloaded") toast.success("Image générée.");
   495→    } catch {
   496→      toast.error("Impossible de générer l’image.");
   497→    } finally {
   498→      setSharePending(false);
   499→    }
   500→  };
   501→
   502→  return (
   503→    <div className="mx-auto w-full max-w-[920px] px-5 py-8 sm:py-10">
   504→      <Breadcrumb
   505→        items={[{ label: "Accueil", href: "accueil" }, { label: "Statistiques" }]}
   506→      />
   507→
   508→      <h1 className="text-3xl font-bold tracking-tight sm:text-4xl">Statistiques</h1>
   509→      <p className="mt-3 max-w-[680px] text-[17px] text-muted-foreground">
   510→        Vos calendriers de concentration, mois après mois et année après année. Vos
   511→        données restent sur votre appareil.
   512→      </p>
   513→
   514→      {!hasData && (
   515→        <div className="mt-6 rounded-3xl border border-dashed p-8 text-center">
   516→          <p className="text-[15px] text-soft">
   517→            Aucune donnée pour l’instant. Terminez votre premier Pomodoro pour voir
   518→            votre calendrier se remplir.
   519→          </p>
   520→        </div>
   521→      )}
   522→
   523→      <div className="mt-8 flex flex-col gap-5">
   524→        {/* ----- KPI row ----- */}
